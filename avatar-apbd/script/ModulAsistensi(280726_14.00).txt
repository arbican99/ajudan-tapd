import React, { useState, useMemo, useEffect } from 'react';
import { supabase } from '../supabaseClient';
import { 
  Save, Trash2, Edit2, CheckCircle2, 
  Layers, Search, FileText, 
  DollarSign, ShoppingBag, CreditCard, AlertCircle, RefreshCw,
  Cpu, Database, Activity, Sparkles, Download, X, AlertTriangle, PlusCircle
} from 'lucide-react';
import * as XLSX from 'xlsx';
import jsPDF from 'jspdf';
import autoTable from 'jspdf-autotable';

const DINKES_KODE = '1.02.0.00.0.00.01.0000';

export default function ModulAsistensi() {
  // === STATE DATA SUPABASE ===
  const [listStatus, setListStatus] = useState([]);
  const [listSkpd, setListSkpd] = useState([]);
  const [listRekening, setListRekening] = useState([]);
  const [listRka, setListRka] = useState([]);
  const [dbAsistensi, setDbAsistensi] = useState([]);
  const [loadingDb, setLoadingDb] = useState(true);

  // STATE KHUSUS SUBUNIT DINAS KESEHATAN
  const [listSubunitDinkes, setListSubunitDinkes] = useState([]);
  const [loadingSubunit, setLoadingSubunit] = useState(false);

  // === FILTER STATES ===
  const [tahun, setTahun] = useState('2026');
  const [kdStatus, setKdStatus] = useState('');
  const [kdSkpd, setKdSkpd] = useState('');
  const [selectedSubunit, setSelectedSubunit] = useState(null);

  // === TAB / PAGE FRAME STATE ===
  const [activeTab, setActiveTab] = useState('pendapatan');

  useEffect(() => {
    fetchSupabaseData();
  }, []);

  // Fetch Data RKA Tanpa Limit
  const fetchAllRkaData = async () => {
    let allData = [];
    let page = 0;
    const pageSize = 1000;
    let fetchMore = true;

    while (fetchMore) {
      const from = page * pageSize;
      const to = from + pageSize - 1;

      const { data, error } = await supabase
        .from('rka')
        .select('*')
        .range(from, to);

      if (error) {
        console.error('Error fetching RKA page', page, error);
        throw error;
      }

      if (data && data.length > 0) {
        allData = [...allData, ...data];
        page++;
        if (data.length < pageSize) fetchMore = false;
      } else {
        fetchMore = false;
      }
    }
    return allData;
  };

  // Fetch Data Asistensi Tanpa Limit
  const fetchAllAsistensiData = async () => {
    let allData = [];
    let page = 0;
    const pageSize = 1000;
    let fetchMore = true;

    while (fetchMore) {
      const from = page * pageSize;
      const to = from + pageSize - 1;

      const { data, error } = await supabase
        .from('asistensi')
        .select('*')
        .order('id', { ascending: false })
        .range(from, to);

      if (error) {
        console.error('Error fetching Asistensi page', page, error);
        throw error;
      }

      if (data && data.length > 0) {
        allData = [...allData, ...data];
        page++;
        if (data.length < pageSize) fetchMore = false;
      } else {
        fetchMore = false;
      }
    }
    return allData;
  };

  const fetchSupabaseData = async () => {
    setLoadingDb(true);
    try {
      const { data: statusData, error: errStatus } = await supabase.from('tblstatus').select('*');
      if (errStatus) throw errStatus;

      const { data: skpdData, error: errSkpd } = await supabase.from('tblskpd').select('*');
      if (errSkpd) throw errSkpd;

      const { data: rekData, error: errRek } = await supabase.from('mrek').select('*');
      if (errRek) throw errRek;

      const rkaData = await fetchAllRkaData();
      const asistensiData = await fetchAllAsistensiData();

      setListStatus(statusData || []);
      setListSkpd(skpdData || []);
      setListRekening(rekData || []);
      setListRka(rkaData || []);
      setDbAsistensi(asistensiData || []);
    } catch (error) {
      console.error('Error loading Supabase:', error.message);
      alert('Gagal memuat data dari Supabase: ' + error.message);
    } finally {
      setLoadingDb(false);
    }
  };

  const fetchAsistensiData = async () => {
    try {
      const asistensiData = await fetchAllAsistensiData();
      setDbAsistensi(asistensiData || []);
    } catch (err) {
      console.error('Error fetching asistensi:', err);
    }
  };

  // FETCH DATA SUBUNIT SPESIFIK APABILA SKPD DINAS KESEHATAN DIPILIH
  useEffect(() => {
    setSelectedSubunit(null);
    if (kdSkpd === DINKES_KODE) {
      const fetchSubunits = async () => {
        setLoadingSubunit(true);
        try {
          const { data, error } = await supabase
            .from('tblskpd')
            .select('kd_subunit, nm_subunit')
            .eq('kd_skpd', DINKES_KODE);

          if (error) throw error;
          setListSubunitDinkes(data || []);
        } catch (err) {
          console.error('Error fetching subunit:', err.message);
        } finally {
          setLoadingSubunit(false);
        }
      };
      fetchSubunits();
    } else {
      setListSubunitDinkes([]);
    }
  }, [kdSkpd]);

  // Daftar SKPD Utama dari tblskpd
  const listSkpdUtama = useMemo(() => {
    const map = new Map();
    listSkpd.forEach(s => {
      const kd = String(s.kd_skpd || s.kdskpd || s.id || '').trim();
      if (!map.has(kd)) map.set(kd, s);
    });
    return Array.from(map.values());
  }, [listSkpd]);

  // Validasi Kelengkapan Filter
  const isFilterComplete = useMemo(() => {
    if (!tahun || !kdStatus || !kdSkpd) return false;
    if (kdSkpd === DINKES_KODE && !selectedSubunit) return false;
    return true;
  }, [tahun, kdStatus, kdSkpd, selectedSubunit]);

  const selectedSkpdObj = useMemo(() => {
    return listSkpdUtama.find(s => String(s.kd_skpd || s.kdskpd) === String(kdSkpd));
  }, [listSkpdUtama, kdSkpd]);

  const selectedStatusObj = useMemo(() => {
    return listStatus.find(s => String(s.kd_status || s.kdstatus || s.id) === String(kdStatus));
  }, [listStatus, kdStatus]);

  const isSubRincianObjek = (r) => {
    const nmlevel = String(r.nmlevel || r.nm_level || '').trim().toUpperCase();
    const level = String(r.level || '').trim();
    return nmlevel === 'SUB RINCIAN OBJEK' || level === '8';
  };

  const rekPendapatanOptions = useMemo(() => {
    return listRekening.filter(r => 
      String(r.kd_rek || r.kdrek || '').startsWith('4') && isSubRincianObjek(r)
    );
  }, [listRekening]);

  const rekBelanjaOptions = useMemo(() => {
    return listRekening.filter(r => 
      String(r.kd_rek || r.kdrek || '').startsWith('5') && isSubRincianObjek(r)
    );
  }, [listRekening]);

  const rekPembiayaanOptions = useMemo(() => {
    return listRekening.filter(r => 
      String(r.kd_rek || r.kdrek || '').startsWith('6') && isSubRincianObjek(r)
    );
  }, [listRekening]);

  // Filter RKA Sub Kegiatan
  const rkaSubgiatOptions = useMemo(() => {
    if (!listRka || listRka.length === 0) return [];
    const uniqueMap = new Map();
    const normalize = (val) => String(val || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase();
    const cleanRaw = (val) => String(val || '').trim();

    const targetStatusName = selectedStatusObj 
      ? cleanRaw(selectedStatusObj.nmstatus || selectedStatusObj.nm_status || selectedStatusObj.status)
      : cleanRaw(kdStatus);

    listRka.forEach((r) => {
      const fitTahun = !tahun || cleanRaw(r.tahun) === cleanRaw(tahun);
      const rkaStatusVal = cleanRaw(r.kdstatus || r.status);
      const fitStatus = !kdStatus || rkaStatusVal === cleanRaw(kdStatus) || rkaStatusVal === targetStatusName;
      const fitSkpd = !kdSkpd || normalize(r.kdskpd) === normalize(kdSkpd);

      let fitSubunit = true;
      if (normalize(kdSkpd) === normalize(DINKES_KODE) && selectedSubunit) {
        const targetSub = normalize(selectedSubunit.kd_subunit || selectedSubunit.kdsubunit);
        fitSubunit = normalize(r.kdsubunit) === targetSub;
      }

      if (fitTahun && fitStatus && fitSkpd && fitSubunit && r.kdsubgiat) {
        const key = cleanRaw(r.kdsubgiat);
        if (!uniqueMap.has(key)) uniqueMap.set(key, r);
      }
    });

    return Array.from(uniqueMap.values());
  }, [listRka, tahun, kdStatus, kdSkpd, selectedSubunit, selectedStatusObj]);

  // Data Asistensi Filtered per SKPD, Status, Tahun & Subunit
  const filteredDbAsistensi = useMemo(() => {
    if (!kdSkpd) return [];
    const normalize = (val) => String(val || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase();

    return dbAsistensi.filter((item) => {
      const fitSkpd = normalize(item.kdskpd) === normalize(kdSkpd);
      const fitTahun = !tahun || String(item.tahun) === String(tahun);
      const fitStatus = !kdStatus || String(item.kdstatus) === String(kdStatus);

      if (!fitSkpd || !fitTahun || !fitStatus) return false;

      if (normalize(kdSkpd) === normalize(DINKES_KODE) && selectedSubunit) {
        const subKode = normalize(selectedSubunit.kd_subunit || selectedSubunit.kdsubunit);
        return normalize(item.kdsubunit) === subKode;
      }

      return true;
    });
  }, [dbAsistensi, kdSkpd, tahun, kdStatus, selectedSubunit]);

  return (
    <div className="min-h-screen bg-[#050811] text-slate-100 p-4 sm:p-6 lg:p-8 font-sans space-y-6 relative overflow-hidden">
      {/* DEKORASI BACKGROUND */}
      <div className="absolute top-0 left-1/4 w-96 h-96 bg-cyan-600/10 rounded-full blur-[120px] pointer-events-none" />
      <div className="absolute bottom-10 right-10 w-[30rem] h-[30rem] bg-indigo-600/10 rounded-full blur-[150px] pointer-events-none" />

      {/* HEADER BAR */}
      <div className="relative z-10 flex flex-col sm:flex-row sm:items-center justify-between gap-4 bg-slate-900/40 px-5 py-4 rounded-2xl border border-cyan-500/20 shadow-xl backdrop-blur-xl">
        <div className="flex items-center gap-3">
          <div className="p-2.5 bg-gradient-to-br from-cyan-500/20 to-indigo-500/20 rounded-xl border border-cyan-500/30 text-cyan-400">
            <Cpu size={22} className="animate-pulse" />
          </div>
          <div>
            <div className="flex items-center gap-2">
              <h1 className="text-base font-extrabold tracking-wider text-transparent bg-clip-text bg-gradient-to-r from-cyan-400 via-teal-300 to-indigo-300 uppercase">
                ASISTENSI APBD
              </h1>
              <span className="px-2 py-0.5 text-[9px] font-mono uppercase bg-cyan-950/80 text-cyan-300 border border-cyan-500/30 rounded-full">
                v2026
              </span>
            </div>
            <p className="text-[11px] text-slate-400 font-mono flex items-center gap-1.5 mt-0.5">
              <Database size={11} className="text-cyan-400" /> Input Hasil Asistensi
            </p>
          </div>
        </div>

        <button
          type="button"
          onClick={fetchSupabaseData}
          disabled={loadingDb}
          className="flex items-center gap-1.5 px-3.5 py-2 rounded-xl bg-slate-900/80 hover:bg-slate-800 text-white border border-slate-700 font-mono text-xs transition-all cursor-pointer disabled:opacity-50"
        >
          <RefreshCw size={13} className={loadingDb ? 'animate-spin text-cyan-400' : 'text-slate-400'} />
          {loadingDb ? 'SYNCING...' : 'RELOAD DB'}
        </button>
      </div>

      {/* FILTER PANEL */}
      <div className="relative z-10 bg-slate-900/30 p-5 rounded-2xl border border-slate-800/80 backdrop-blur-xl shadow-2xl space-y-4">
        <div className="flex items-center gap-2 pb-2 border-b border-slate-800 text-[11px] font-mono text-cyan-400 uppercase tracking-widest">
          <Sparkles size={13} /> Filter Data Inputan
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
          <div className="space-y-1.5">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">1. Tahun Anggaran</label>
            <select
              value={tahun}
              onChange={(e) => setTahun(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-400 rounded-xl px-3.5 py-2.5 text-xs font-mono text-white focus:outline-none"
            >
              <option value="2025">2025</option>
              <option value="2026">2026</option>
              <option value="2027">2027</option>
            </select>
          </div>

          <div className="space-y-1.5">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">2. Tahapan APBD</label>
            <select
              value={kdStatus}
              onChange={(e) => setKdStatus(e.target.value)}
              disabled={loadingDb}
              className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-400 rounded-xl px-3.5 py-2.5 text-xs font-mono text-white focus:outline-none disabled:opacity-50"
            >
              <option value="">-- PILIH TAHAPAN APBD --</option>
              {listStatus.map((s, idx) => {
                const valKode = String(s.kdstatus || s.kd_status || s.id || '').trim();
                const valNama = s.nmstatus || s.nm_status || s.status || valKode;
                return (
                  <option key={idx} value={valKode}>[{valKode}] {valNama}</option>
                );
              })}
            </select>
          </div>

          <div className="space-y-1.5">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">3. Perangkat Daerah</label>
            <select
              value={kdSkpd}
              onChange={(e) => setKdSkpd(e.target.value)}
              disabled={loadingDb}
              className="w-full bg-slate-950 border border-slate-800 focus:border-cyan-400 rounded-xl px-3.5 py-2.5 text-xs font-mono text-white focus:outline-none disabled:opacity-50"
            >
              <option value="">-- PILIH SKPD / PERANGKAT DAERAH --</option>
              {listSkpdUtama.map((skpd, idx) => {
                const kdS = skpd.kdskpd || skpd.kd_skpd || skpd.id;
                const nmS = skpd.nmskpd || skpd.nm_skpd || skpd.nama_skpd;
                return (
                  <option key={idx} value={kdS}>[{kdS}] {nmS}</option>
                );
              })}
            </select>
          </div>
        </div>

        {kdSkpd === DINKES_KODE && (
          <div className="mt-2 space-y-1.5 pt-2 border-t border-slate-800/60">
            <div className="flex items-center justify-between">
              <label className="text-[10px] font-mono font-bold text-cyan-400 uppercase flex items-center gap-1.5">
                <Layers size={13} /> 4. Subunit Dinas Kesehatan (Wajib Dipilih)
              </label>
              {loadingSubunit && <span className="text-[10px] font-mono text-slate-400">Loading Subunit...</span>}
            </div>
            <select
              value={selectedSubunit ? (selectedSubunit.kd_subunit || selectedSubunit.kdsubunit) : ''}
              onChange={(e) => {
                const val = e.target.value;
                const found = listSubunitDinkes.find(sub => String(sub.kd_subunit || sub.kdsubunit) === String(val));
                setSelectedSubunit(found || null);
              }}
              disabled={loadingSubunit}
              className="w-full bg-slate-950 border border-cyan-500/50 focus:border-cyan-400 rounded-xl px-3.5 py-2.5 text-xs font-mono text-white focus:outline-none disabled:opacity-50"
            >
              <option value="">-- PILIH SUBUNIT DINAS KESEHATAN --</option>
              {listSubunitDinkes.map((sub, idx) => {
                const kodeSub = sub.kd_subunit || sub.kdsubunit;
                const namaSub = sub.nm_subunit || sub.nmsubunit;
                return (
                  <option key={idx} value={kodeSub}>
                    [{kodeSub}] {namaSub}
                  </option>
                );
              })}
            </select>
          </div>
        )}
      </div>

      {/* TAB MENU & CONTENT FRAME */}
      {!isFilterComplete ? (
        <div className="p-12 text-center bg-slate-900/20 rounded-2xl border border-dashed border-slate-800 space-y-2">
          <AlertCircle size={36} className="mx-auto text-amber-400 animate-bounce" />
          <h3 className="text-sm font-bold text-slate-200 tracking-wide font-mono">FILTER INCOMPLETE</h3>
          <p className="text-xs text-slate-400 max-w-md mx-auto">
            Silakan tentukan Tahun Anggaran, Status APBD, dan SKPD {kdSkpd === DINKES_KODE ? '(serta Subunit Wajib)' : ''} untuk mengaktifkan modul asistensi.
          </p>
        </div>
      ) : (
        <div className="space-y-4">
          <div className="flex flex-wrap gap-2 border-b border-slate-800 pb-3">
            {[
              { id: 'pendapatan', label: 'Pendapatan Daerah', icon: ShoppingBag ,color: 'text-emerald-400' },
              { id: 'belanja', label: 'Belanja', icon: ShoppingBag, color: 'text-cyan-400' },
              { id: 'pembiayaan', label: 'Pembiayaan', icon: CreditCard, color: 'text-amber-400' },
              { id: 'laporan', label: 'Laporan', icon: FileText, color: 'text-indigo-400' },
            ].map((tab) => {
              const Icon = tab.icon;
              const isActive = activeTab === tab.id;
              return (
                <button
                  key={tab.id}
                  onClick={() => setActiveTab(tab.id)}
                  className={`flex items-center gap-2 px-4 py-2.5 rounded-xl font-mono text-xs uppercase tracking-wider transition-all cursor-pointer ${
                    isActive
                      ? 'bg-slate-800 text-white border border-cyan-500/50 shadow-md'
                      : 'bg-slate-900/40 text-slate-400 hover:bg-slate-900 hover:text-white border border-slate-800'
                  }`}
                >
                  <Icon size={15} className={tab.color} />
                  {tab.label}
                </button>
              );
            })}
          </div>

          <div className="bg-slate-900/40 p-5 rounded-2xl border border-slate-800 backdrop-blur-xl shadow-2xl">
            {activeTab === 'pendapatan' && (
              <InputTableFrame
                type="pendapatan"
                title="Input Asistensi Pendapatan"
                rekPrefix="4"
                rekOptions={rekPendapatanOptions}
                dbAsistensi={filteredDbAsistensi.filter(d => d.kdrek.startsWith('4'))}
                fetchAsistensiData={fetchAsistensiData}
                filterParams={{ tahun, kdStatus, kdSkpd, selectedSkpdObj, selectedStatusObj, selectedSubunit }}
                hasSubgiat={false}
              />
            )}

            {activeTab === 'belanja' && (
              <InputTableFrame
                type="belanja"
                title="Input Asistensi Belanja"
                rekPrefix="5"
                rekOptions={rekBelanjaOptions}
                rkaSubgiatOptions={rkaSubgiatOptions}
                dbAsistensi={filteredDbAsistensi.filter(d => d.kdrek.startsWith('5'))}
                fetchAsistensiData={fetchAsistensiData}
                filterParams={{ tahun, kdStatus, kdSkpd, selectedSkpdObj, selectedStatusObj, selectedSubunit }}
                hasSubgiat={true}
              />
            )}

            {activeTab === 'pembiayaan' && (
              <InputTableFrame
                type="pembiayaan"
                title="Input Asistensi Pembiayaan"
                rekPrefix="6"
                rekOptions={rekPembiayaanOptions}
                dbAsistensi={filteredDbAsistensi.filter(d => d.kdrek.startsWith('6'))}
                fetchAsistensiData={fetchAsistensiData}
                filterParams={{ tahun, kdStatus, kdSkpd, selectedSkpdObj, selectedStatusObj, selectedSubunit }}
                hasSubgiat={false}
              />
            )}

            {activeTab === 'laporan' && (
              <LaporanFrame 
                dbAsistensi={dbAsistensi} 
                selectedKdSkpd={kdSkpd}
                selectedSkpdObj={selectedSkpdObj}
                selectedStatusObj={selectedStatusObj}
                selectedSubunit={selectedSubunit}
                listSubunitDinkes={listSubunitDinkes}
                tahun={tahun}
                kdStatus={kdStatus}
              />
            )}
          </div>
        </div>
      )}
    </div>
  );
}

// === COMBOBOX SUBKEGIATAN SEARCHABLE ===
function SelectSubgiatSearchable({ rkaSubgiatOptions, value, onChange }) {
  const [isOpen, setIsOpen] = useState(false);
  const [searchSub, setSearchSub] = useState('');

  const selectedSub = useMemo(() => {
    return rkaSubgiatOptions.find(r => String(r.kdsubgiat || r.kd_subgiat) === String(value));
  }, [rkaSubgiatOptions, value]);

  const filteredOptions = useMemo(() => {
    if (!searchSub.trim()) return rkaSubgiatOptions;
    const term = searchSub.toLowerCase();
    return rkaSubgiatOptions.filter(r => {
      const kd = String(r.kdsubgiat || r.kd_subgiat || '').toLowerCase();
      const nm = String(r.nmsubgiat || r.nm_subgiat || '').toLowerCase();
      return kd.includes(term) || nm.includes(term);
    });
  }, [rkaSubgiatOptions, searchSub]);

  return (
    <div className="relative w-full">
      <div
        onClick={() => setIsOpen(!isOpen)}
        className="w-full bg-slate-900 border border-slate-800 focus-within:border-cyan-400 rounded-xl px-3 py-2 text-xs font-mono text-white flex items-center justify-between cursor-pointer hover:border-cyan-500/50 min-h-[38px]"
      >
        <div className="truncate pr-2">
          {selectedSub ? (
            <span>
              <strong className="text-cyan-400">[{selectedSub.kdsubgiat || selectedSub.kd_subgiat}]</strong>{' '}
              {selectedSub.nmsubgiat || selectedSub.nm_subgiat}
            </span>
          ) : (
            <span className="text-slate-500">-- Pilih & Cari Sub Kegiatan --</span>
          )}
        </div>
        <Search size={13} className="text-slate-400 shrink-0" />
      </div>

      {isOpen && (
        <div className="absolute z-50 left-0 top-full mt-1 w-full bg-slate-950 border border-cyan-500/40 rounded-xl shadow-2xl p-2 space-y-2 max-h-64 overflow-hidden flex flex-col">
          <input
            type="text"
            placeholder="Ketik kode / nama sub kegiatan..."
            value={searchSub}
            onChange={(e) => setSearchSub(e.target.value)}
            className="w-full bg-slate-900 border border-slate-800 rounded-lg px-3 py-1.5 text-xs text-white placeholder-slate-500 focus:outline-none focus:border-cyan-400 font-mono"
            autoFocus
          />

          <div className="overflow-y-auto max-h-48 space-y-1 pr-1">
            {filteredOptions.length === 0 ? (
              <div className="p-3 text-center text-[11px] text-slate-500 font-mono">
                Sub Kegiatan tidak ditemukan.
              </div>
            ) : (
              filteredOptions.map((r, idx) => {
                const kdS = r.kdsubgiat || r.kd_subgiat;
                const nmS = r.nmsubgiat || r.nm_subgiat || '';
                const isSelected = String(value) === String(kdS);

                return (
                  <div
                    key={idx}
                    onClick={() => {
                      onChange(kdS, nmS);
                      setIsOpen(false);
                      setSearchSub('');
                    }}
                    className={`p-2 rounded-lg cursor-pointer text-xs font-mono flex items-center gap-2 ${
                      isSelected ? 'bg-cyan-950 border border-cyan-500/40 text-cyan-300' : 'hover:bg-slate-900 text-slate-200'
                    }`}
                  >
                    <span className="text-cyan-400 font-bold text-[10px] shrink-0">[{kdS}]</span>
                    <span className="text-white text-[11px] font-sans truncate">{nmS}</span>
                  </div>
                );
              })
            )}
          </div>
        </div>
      )}
    </div>
  );
}

// === COMBOBOX REKENING SEARCHABLE (KODE DAN NAMA SEJAJAR) ===
function SelectRekeningSearchable({ rekOptions, value, onChange }) {
  const [isOpen, setIsOpen] = useState(false);
  const [searchRek, setSearchRek] = useState('');

  const selectedRek = useMemo(() => {
    return rekOptions.find(r => String(r.kdrek || r.kd_rek) === String(value));
  }, [rekOptions, value]);

  const filteredOptions = useMemo(() => {
    if (!searchRek.trim()) return rekOptions;
    const term = searchRek.toLowerCase();
    return rekOptions.filter(r => {
      const kd = String(r.kdrek || r.kd_rek || '').toLowerCase();
      const nm = String(r.nmrek || r.nm_rek || r.nama_rekening || r.uraian || '').toLowerCase();
      return kd.includes(term) || nm.includes(term);
    });
  }, [rekOptions, searchRek]);

  return (
    <div className="relative w-full">
      <div
        onClick={() => setIsOpen(!isOpen)}
        className="w-full bg-slate-900 border border-slate-800 focus-within:border-cyan-400 rounded-xl px-3 py-2 text-xs font-mono text-white flex items-center justify-between cursor-pointer hover:border-cyan-500/50 min-h-[38px]"
      >
        <div className="truncate pr-2">
          {selectedRek ? (
            <span>
              <strong className="text-cyan-400">[{selectedRek.kdrek || selectedRek.kd_rek}]</strong>{' '}
              {selectedRek.nmrek || selectedRek.nm_rek || selectedRek.nama_rekening || selectedRek.uraian}
            </span>
          ) : (
            <span className="text-slate-500">-- Pilih & Cari Rekening --</span>
          )}
        </div>
        <Search size={13} className="text-slate-400 shrink-0" />
      </div>

      {isOpen && (
        <div className="absolute z-50 left-0 top-full mt-1 w-full bg-slate-950 border border-cyan-500/40 rounded-xl shadow-2xl p-2 space-y-2 max-h-64 overflow-hidden flex flex-col">
          <input
            type="text"
            placeholder="Ketik kode / nama rekening..."
            value={searchRek}
            onChange={(e) => setSearchRek(e.target.value)}
            className="w-full bg-slate-900 border border-slate-800 rounded-lg px-3 py-1.5 text-xs text-white placeholder-slate-500 focus:outline-none focus:border-cyan-400 font-mono"
            autoFocus
          />

          <div className="overflow-y-auto max-h-48 space-y-1 pr-1">
            {filteredOptions.length === 0 ? (
              <div className="p-3 text-center text-[11px] text-slate-500 font-mono">
                Rekening tidak ditemukan.
              </div>
            ) : (
              filteredOptions.map((r, idx) => {
                const kdR = r.kdrek || r.kd_rek;
                const nmR = r.nmrek || r.nm_rek || r.nama_rekening || r.uraian || '';
                const isSelected = String(value) === String(kdR);

                return (
                  <div
                    key={idx}
                    onClick={() => {
                      onChange(kdR, nmR);
                      setIsOpen(false);
                      setSearchRek('');
                    }}
                    className={`p-2 rounded-lg cursor-pointer text-xs font-mono flex items-center gap-2 ${
                      isSelected ? 'bg-cyan-950 border border-cyan-500/40 text-cyan-300' : 'hover:bg-slate-900 text-slate-200'
                    }`}
                  >
                    <span className="text-cyan-400 font-bold text-[10px] shrink-0">[{kdR}]</span>
                    <span className="text-white text-[11px] font-sans truncate">{nmR}</span>
                  </div>
                );
              })
            )}
          </div>
        </div>
      )}
    </div>
  );
}

// === KOMPONEN FORM INPUTAN & TABEL SUPABASE DIRECT ===
function InputTableFrame({ title, rekPrefix, rekOptions, rkaSubgiatOptions = [], dbAsistensi, fetchAsistensiData, filterParams, hasSubgiat }) {
  const [editingId, setEditingId] = useState(null);
  const [saving, setSaving] = useState(false);
  const [deletingByFilter, setDeletingByFilter] = useState(false);

  // Local Table State menampung data asistensi sementara
  const [localAsistensi, setLocalAsistensi] = useState([]);

  useEffect(() => {
    setLocalAsistensi(dbAsistensi);
  }, [dbAsistensi]);

  const [form, setForm] = useState({
    kdsubgiat: '',
    nmsubgiat: '',
    kdrek: '',
    nmrek: '',
    usulan: 'Penambahan',
    jumlah: '',
    keterangan: '',
    setuju: 'Disetujui'
  });

  const formatRupiah = (val) => {
    if (!val && val !== 0) return '';
    const cleanNum = String(val).replace(/\D/g, '');
    return cleanNum ? Number(cleanNum).toLocaleString('id-ID') : '';
  };

  const handleJumlahChange = (e) => {
    const rawVal = e.target.value.replace(/\D/g, '');
    setForm(prev => ({ ...prev, jumlah: rawVal }));
  };

  const resetForm = () => {
    setForm({
      kdsubgiat: '',
      nmsubgiat: '',
      kdrek: '',
      nmrek: '',
      usulan: 'Penambahan',
      jumlah: '',
      keterangan: '',
      setuju: 'Disetujui'
    });
    setEditingId(null);
  };

  // 1. Simpan Ke Tabel Asistensi (Local State UI)
  const handleSaveToTable = () => {
    if (!form.kdrek) {
      alert('Pilih Rekening terlebih dahulu!');
      return;
    }
    if (hasSubgiat && !form.kdsubgiat) {
      alert('Pilih Sub Kegiatan terlebih dahulu!');
      return;
    }
    if (!form.jumlah || isNaN(form.jumlah) || Number(form.jumlah) <= 0) {
      alert('Isi Jumlah nominal angka yang valid!');
      return;
    }

    const { tahun, kdStatus, kdSkpd, selectedSkpdObj, selectedStatusObj, selectedSubunit } = filterParams;
    const kdSubunitVal = selectedSubunit ? (selectedSubunit.kd_subunit || selectedSubunit.kdsubunit) : '-';
    const nmSubunitVal = selectedSubunit ? (selectedSubunit.nm_subunit || selectedSubunit.nmsubunit) : '-';

    const newRecord = {
      id: editingId || `temp_${Date.now()}`,
      tahapan: '1',
      tahun: String(tahun),
      status: selectedStatusObj?.nm_status || selectedStatusObj?.status || selectedStatusObj?.nmstatus || '',
      kdstatus: String(kdStatus),
      kdskpd: String(kdSkpd),
      nmskpd: selectedSkpdObj?.nm_skpd || selectedSkpdObj?.nmskpd || '',
      kdsubunit: kdSubunitVal,
      nmsubunit: nmSubunitVal,
      kdsubgiat: form.kdsubgiat || '-',
      nmsubgiat: form.nmsubgiat || '-',
      kdrek: String(form.kdrek),
      nmrek: String(form.nmrek),
      usulan: String(form.usulan),
      keterangan: String(form.keterangan || '-'),
      setuju: String(form.setuju || 'Disetujui'),
      jumlah: Number(form.jumlah) || 0
    };

    if (editingId) {
      setLocalAsistensi(prev => prev.map(item => item.id === editingId ? newRecord : item));
    } else {
      const existIdx = localAsistensi.findIndex(item => 
        String(item.kdsubgiat) === String(newRecord.kdsubgiat) && 
        String(item.kdrek) === String(newRecord.kdrek)
      );

      if (existIdx >= 0) {
        setLocalAsistensi(prev => prev.map((item, idx) => idx === existIdx ? newRecord : item));
      } else {
        setLocalAsistensi(prev => [newRecord, ...prev]);
      }
    }

    resetForm();
  };

  // 2. Simpan Seluruh Data Tabel Ke Supabase
  const handleSyncToSupabase = async () => {
    if (localAsistensi.length === 0) {
      alert('Tabel asistensi masih kosong! Tambahkan data terlebih dahulu.');
      return;
    }

    setSaving(true);
    try {
      for (const item of localAsistensi) {
        const payload = {
          tahapan: item.tahapan,
          tahun: item.tahun,
          status: item.status,
          kdstatus: item.kdstatus,
          kdskpd: item.kdskpd,
          nmskpd: item.nmskpd,
          kdsubunit: item.kdsubunit,
          nmsubunit: item.nmsubunit,
          kdsubgiat: item.kdsubgiat,
          nmsubgiat: item.nmsubgiat,
          kdrek: item.kdrek,
          nmrek: item.nmrek,
          usulan: item.usulan,
          keterangan: item.keterangan,
          setuju: item.setuju,
          jumlah: item.jumlah
        };

        if (typeof item.id === 'number') {
          const { error } = await supabase.from('asistensi').update(payload).eq('id', item.id);
          if (error) throw error;
        } else {
          const { data: existingData, error: checkErr } = await supabase
            .from('asistensi')
            .select('id')
            .eq('tahun', item.tahun)
            .eq('kdstatus', item.kdstatus)
            .eq('kdskpd', item.kdskpd)
            .eq('kdsubunit', item.kdsubunit)
            .eq('kdsubgiat', item.kdsubgiat)
            .eq('kdrek', item.kdrek);

          if (checkErr) throw checkErr;

          if (existingData && existingData.length > 0) {
            const { error: updateErr } = await supabase.from('asistensi').update(payload).eq('id', existingData[0].id);
            if (updateErr) throw updateErr;
          } else {
            const { error: insertErr } = await supabase.from('asistensi').insert([payload]);
            if (insertErr) throw insertErr;
          }
        }
      }

      alert('Berhasil menyimpan/memperbarui seluruh data tabel asistensi ke Supabase!');
      await fetchAsistensiData();
    } catch (err) {
      console.error('Gagal sinkronisasi Supabase:', err);
      alert('Error sinkronisasi Supabase: ' + err.message);
    } finally {
      setSaving(false);
    }
  };

  const handleDeleteByFilter = async () => {
    const { tahun, kdStatus, kdSkpd, selectedSkpdObj, selectedSubunit } = filterParams;
    const kdSubunitVal = selectedSubunit ? (selectedSubunit.kd_subunit || selectedSubunit.kdsubunit) : '-';
    
    const namaSkpd = selectedSkpdObj?.nm_skpd || selectedSkpdObj?.nmskpd || kdSkpd;
    const namaSubunit = selectedSubunit ? ` Subunit: ${selectedSubunit.nm_subunit || selectedSubunit.nmsubunit}` : '';

    const confirmMsg = `APAKAH ANDA YAKIN INGIN MENGHAPUS SELURUH DATA SAMA SESUAI FILTER?\n\n` +
      `• Tahun: ${tahun}\n` +
      `• Tahapan/Status: ${kdStatus}\n` +
      `• SKPD: [${kdSkpd}] ${namaSkpd}${namaSubunit}\n` +
      `• Jenis: Rekening kelompok '${rekPrefix}' (${title})\n` +
      `• Total Data Ditemukan: ${localAsistensi.length} Record\n\n` +
      `Tindakan ini tidak dapat dibatalkan!`;

    if (!confirm(confirmMsg)) return;

    setDeletingByFilter(true);
    try {
      let query = supabase.from('asistensi')
        .delete()
        .eq('tahun', String(tahun))
        .eq('kdstatus', String(kdStatus))
        .eq('kdskpd', String(kdSkpd))
        .like('kdrek', `${rekPrefix}%`);

      if (selectedSubunit) {
        query = query.eq('kdsubunit', kdSubunitVal);
      }

      const { error } = await query;
      if (error) throw error;

      alert(`Berhasil menghapus seluruh data asistensi sesuai filter aktif!`);
      resetForm();
      await fetchAsistensiData();
    } catch (err) {
      console.error('Gagal hapus sesuai filter:', err);
      alert('Gagal menghapus data: ' + err.message);
    } finally {
      setDeletingByFilter(false);
    }
  };

  const handleEdit = (item) => {
    setEditingId(item.id);
    setForm({
      kdsubgiat: item.kdsubgiat !== '-' ? item.kdsubgiat : '',
      nmsubgiat: item.nmsubgiat !== '-' ? item.nmsubgiat : '',
      kdrek: item.kdrek,
      nmrek: item.nmrek,
      usulan: item.usulan,
      jumlah: String(item.jumlah),
      keterangan: item.keterangan !== '-' ? item.keterangan : '',
      setuju: item.setuju
    });
  };

  const handleDeleteLocal = async (item) => {
    if (!confirm('Apakah Anda yakin ingin menghapus data asistensi ini?')) return;
    
    if (typeof item.id === 'number') {
      try {
        const { error } = await supabase.from('asistensi').delete().eq('id', item.id);
        if (error) throw error;
        alert('Data berhasil dihapus dari Supabase!');
        await fetchAsistensiData();
      } catch (err) {
        alert('Gagal menghapus data: ' + err.message);
      }
    } else {
      setLocalAsistensi(prev => prev.filter(i => i.id !== item.id));
    }
  };

  return (
    <div className="space-y-5">
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <h2 className="text-xs font-mono font-bold text-cyan-400 uppercase tracking-widest flex items-center gap-2">
          <Activity size={15} /> {title}
        </h2>

        {editingId && (
          <span className="text-xs font-mono text-amber-400 font-bold bg-amber-950/80 px-2.5 py-1 rounded-lg border border-amber-800 flex items-center gap-1.5">
            <Edit2 size={12} /> Mode Edit Data (ID: {editingId})
            <button onClick={resetForm} className="ml-2 text-slate-400 hover:text-white cursor-pointer"><X size={12} /></button>
          </span>
        )}
      </div>

      {/* FORM INPUTAN */}
      <div className="bg-slate-950/80 p-4 rounded-xl border border-slate-800 shadow-inner space-y-3">
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-3">
          
          {hasSubgiat && (
            <div className="space-y-1 md:col-span-2">
              <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Sub Kegiatan</label>
              <SelectSubgiatSearchable
                rkaSubgiatOptions={rkaSubgiatOptions}
                value={form.kdsubgiat}
                onChange={(kd, nm) => setForm(prev => ({ ...prev, kdsubgiat: kd, nmsubgiat: nm }))}
              />
            </div>
          )}

          <div className="space-y-1 md:col-span-2">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Kode Rekening</label>
            <SelectRekeningSearchable
              rekOptions={rekOptions}
              value={form.kdrek}
              onChange={(kd, nm) => setForm(prev => ({ ...prev, kdrek: kd, nmrek: nm }))}
            />
          </div>

          <div className="space-y-1">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Jenis Usulan</label>
            <select
              value={form.usulan}
              onChange={(e) => setForm({ ...form, usulan: e.target.value })}
              className="w-full bg-slate-900 border border-slate-800 focus:border-cyan-400 rounded-xl px-3 py-2 text-xs font-mono text-white focus:outline-none"
            >
              <option value="Penambahan">Penambahan</option>
              <option value="Pengurangan">Pengurangan</option>
            </select>
          </div>

          <div className="space-y-1">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Jumlah Nilai (Rp)</label>
            <div className="relative">
              <span className="absolute left-3 top-2 text-xs font-mono text-cyan-400 font-bold">Rp</span>
              <input
                type="text"
                value={formatRupiah(form.jumlah)}
                onChange={handleJumlahChange}
                placeholder="0"
                className="w-full bg-slate-900 border border-slate-800 focus:border-cyan-400 rounded-xl pl-9 pr-3 py-2 text-xs font-mono text-white font-bold focus:outline-none"
              />
            </div>
          </div>

          <div className="space-y-1">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Status</label>
            <select
              value={form.setuju}
              onChange={(e) => setForm({ ...form, setuju: e.target.value })}
              className="w-full bg-slate-900 border border-slate-800 focus:border-cyan-400 rounded-xl px-3 py-2 text-xs font-mono text-white focus:outline-none"
            >
              <option value="Disetujui">Disetujui</option>
              <option value="Tidak Disetujui">Tidak Disetujui</option>
            </select>
          </div>

          <div className="space-y-1 md:col-span-3">
            <label className="text-[10px] font-mono font-bold text-slate-300 uppercase">Keterangan Asistensi Catatan</label>
            <textarea
              rows={2}
              value={form.keterangan}
              onChange={(e) => setForm({ ...form, keterangan: e.target.value })}
              placeholder="Catatan hasil asistensi..."
              className="w-full bg-slate-900 border border-slate-800 focus:border-cyan-400 rounded-xl px-3 py-2 text-xs text-white focus:outline-none font-sans resize-y"
            />
          </div>

          {/* TOMBOL SIMPAN KE TABEL ASISTENSI (LOKAL) */}
          <div className="flex items-center md:col-span-2 lg:col-span-4 pt-2 border-t border-slate-800/80">
            <button
              type="button"
              onClick={handleSaveToTable}
              className="w-full py-2.5 px-4 rounded-xl bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-500 hover:to-indigo-500 text-white font-extrabold text-xs uppercase tracking-wider transition-all flex items-center justify-center gap-2 cursor-pointer shadow-md"
            >
              <PlusCircle size={15} />
              {editingId ? 'UPDATE DI TABEL ASISTENSI' : 'SIMPAN KE TABEL ASISTENSI'}
            </button>
          </div>

        </div>
      </div>

      {/* TABEL DATA ASISTENSI */}
      <div className="space-y-3">
        <div className="flex items-center justify-between text-xs font-mono font-bold text-amber-400 uppercase">
          <span>Tabel Data Asistensi ({localAsistensi.length} Records)</span>
        </div>

        <div className="overflow-x-auto rounded-xl border border-slate-800">
          <table className="w-full text-left text-xs">
            <thead className="bg-slate-950 text-cyan-400 font-mono uppercase border-b border-slate-800 tracking-wider">
              <tr>
                <th className="p-3 text-center">No</th>
                {hasSubgiat && <th className="p-3">Sub Kegiatan</th>}
                <th className="p-3">Kode & Nama Rekening</th>
                <th className="p-3">Usulan</th>
                <th className="p-3 text-right">Jumlah (Rp)</th>
                <th className="p-3 min-w-[220px]">Keterangan</th>
                <th className="p-3 text-center">Status</th>
                <th className="p-3 text-center">Aksi</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-800/60 font-sans bg-slate-950/40">
              {localAsistensi.length === 0 ? (

              ) : (

                    <td className="p-3 text-center font-mono text-slate-400">{idx + 1}</td>
                    {hasSubgiat && (
                      <td className="p-3 max-w-xs">
                        <div className="font-mono text-cyan-300 text-[10px]">{item.kdsubgiat}</div>
                        <div className="text-white truncate text-[11px] font-medium">{item.nmsubgiat}</div>
                      </td>
                    )}
                    <td className="p-3 max-w-xs">
                      <div className="font-mono text-cyan-300 text-[10px]">{item.kdrek}</div>
                      <div className="text-white text-[11px] font-medium">{item.nmrek}</div>
                    </td>
                    <td className="p-3">
                      <span className={`px-2 py-0.5 rounded text-[10px] font-mono font-bold ${
                        item.usulan === 'Penambahan' ? 'bg-emerald-950 text-emerald-300 border border-emerald-800' : 'bg-amber-950 text-amber-300 border border-amber-800'
                      }`}>
                        {item.usulan}
                      </span>
                    </td>
                    <td className="p-3 text-right font-mono text-amber-300 font-bold">
                      {Number(item.jumlah || 0).toLocaleString('id-ID')}
                    </td>
                    <td className="p-3 text-slate-300 text-[11px] min-w-[220px] whitespace-pre-wrap">
                      {item.keterangan || '-'}
                    </td>
                    <td className="p-3 text-center">
                      <span className={`px-2 py-0.5 rounded text-[10px] font-mono font-bold ${
                        item.setuju === 'Disetujui' ? 'bg-cyan-950 text-cyan-300 border border-cyan-800' : 'bg-rose-950 text-rose-300 border border-rose-800'
                      }`}>
                        {item.setuju}
                      </span>
                    </td>
                    <td className="p-3 text-center space-x-1">
                      <button onClick={() => handleEdit(item)} className="text-cyan-400 hover:text-cyan-300 p-1 cursor-pointer" title="Edit Data">
                        <Edit2 size={13} />
                      </button>
                      <button onClick={() => handleDeleteLocal(item)} className="text-rose-400 hover:text-rose-300 p-1 cursor-pointer" title="Hapus Data">
                        <Trash2 size={13} />
                      </button>
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>

        {/* TOMBOL AKSI DI BAWAH TABEL ASISTENSI */}
        <div className="flex flex-col sm:flex-row items-center gap-2.5 pt-3">
          <button
            type="button"
            onClick={handleSyncToSupabase}
            disabled={saving || localAsistensi.length === 0}
            className="flex-1 w-full py-2.5 px-4 rounded-xl bg-gradient-to-r from-cyan-500 to-emerald-500 hover:from-cyan-400 hover:to-emerald-400 text-slate-950 font-extrabold text-xs uppercase tracking-wider transition-all flex items-center justify-center gap-2 cursor-pointer shadow-md disabled:opacity-50"
          >
            <Save size={14} />
            {saving ? 'SAVING TO DATABASE...' : 'SIMPAN DATA'}
          </button>

          <button
            type="button"
            onClick={handleDeleteByFilter}
            disabled={deletingByFilter || localAsistensi.length === 0}
            className="flex-1 w-full py-2.5 px-4 rounded-xl bg-rose-950/80 hover:bg-rose-900 text-rose-300 border border-rose-800/80 font-extrabold text-xs uppercase tracking-wider transition-all flex items-center justify-center gap-2 cursor-pointer shadow-md disabled:opacity-40"
            title="Hapus seluruh data yang cocok dengan parameter filter aktif saat ini"
          >
            <Trash2 size={14} className={deletingByFilter ? 'animate-bounce' : ''} />
            {deletingByFilter ? 'DELETING...' : `HAPUS SEMUA DATA DI TABEL ASISTENSI (${localAsistensi.length})`}
          </button>
        </div>
      </div>
    </div>
  );
}

// === PAGE FRAME LAPORAN ===
function LaporanFrame({ dbAsistensi, selectedKdSkpd, selectedSkpdObj, selectedStatusObj, listSubunitDinkes, tahun, kdStatus }) {
  const [downloadingPdf, setDownloadingPdf] = useState(false);

  const normalize = (val) => String(val || '').replace(/[^a-zA-Z0-9]/g, '').toLowerCase();

  const formatAngkaIndo = (val) => {
    if (!val && val !== 0) return '';
    return Number(val).toLocaleString('id-ID', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  };

  // Filter Data Asistensi Sesuai SKPD, Tahun, dan Tahapan APBD yang dipilih
  const filteredData = useMemo(() => {
    if (!selectedKdSkpd) return [];
    return dbAsistensi.filter(d => 
      normalize(d.kdskpd) === normalize(selectedKdSkpd) &&
      (!tahun || String(d.tahun) === String(tahun)) &&
      (!kdStatus || String(d.kdstatus) === String(kdStatus))
    );
  }, [dbAsistensi, selectedKdSkpd, tahun, kdStatus]);

  // Fungsi Pembentuk Struktur Laporan (Hanya memproses data yang ADA di tabel asistensi)
  const buildReportStructure = (dataItems) => {
    const pendapatan = dataItems.filter(d => d.kdrek && String(d.kdrek).startsWith('4'));
    const belanja = dataItems.filter(d => d.kdrek && String(d.kdrek).startsWith('5'));
    const pembiayaan = dataItems.filter(d => d.kdrek && String(d.kdrek).startsWith('6'));

    const subgiatMap = new Map();
    belanja.forEach(item => {
      const subKey = item.kdsubgiat || 'LAINNYA';
      if (!subgiatMap.has(subKey)) {
        subgiatMap.set(subKey, {
          kdsubgiat: subKey,
          nmsubgiat: item.nmsubgiat || '-',
          items: []
        });
      }
      subgiatMap.get(subKey).items.push(item);
    });

    const totalPenambahan = dataItems.reduce((acc, curr) => acc + (curr.usulan === 'Penambahan' ? Number(curr.jumlah || 0) : 0), 0);
    const totalPengurangan = dataItems.reduce((acc, curr) => acc + (curr.usulan === 'Pengurangan' ? Number(curr.jumlah || 0) : 0), 0);

    return {
      pendapatan,
      belanjaSubgiat: Array.from(subgiatMap.values()),
      pembiayaan,
      totalPenambahan,
      totalPengurangan
    };
  };

  const handleDownloadPDF = async () => {
    setDownloadingPdf(true);
    try {
      const doc = new jsPDF({
        orientation: 'portrait',
        unit: 'mm',
        format: 'a4'
      });

      const pageWidth = doc.internal.pageSize.getWidth();

      doc.setFont('helvetica', 'bold');
      doc.setFontSize(11);
      doc.text('LAPORAN HASIL ASISTENSI APBD', pageWidth / 2, 12, { align: 'center' });
      
      doc.setFontSize(8);
      const namaSkpd = (selectedSkpdObj?.nm_skpd || selectedSkpdObj?.nmskpd || '-').toUpperCase();
      const namaStatus = (selectedStatusObj?.nm_status || selectedStatusObj?.status || selectedStatusObj?.nmstatus || '-').toUpperCase();
      
      doc.text(`SKPD: ${namaSkpd}`, 10, 18);
      doc.text(`STATUS: ${namaStatus} (TAHUN ${tahun})`, 10, 22);

      const tableRows = [];

      const appendStructureToRows = (struct) => {
        let hasDataInStruct = false;

        // 1. PENDAPATAN
        if (struct.pendapatan.length > 0) {
          hasDataInStruct = true;
          const totPenambahanPendapatan = struct.pendapatan.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
          const totPenguranganPendapatan = struct.pendapatan.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);
          
          tableRows.push(['1', '1', 'PENDAPATAN DAERAH', '', totPenambahanPendapatan ? formatAngkaIndo(totPenambahanPendapatan) : '', totPenguranganPendapatan ? formatAngkaIndo(totPenguranganPendapatan) : '', '']);

          struct.pendapatan.forEach(p => {
            tableRows.push([
              '',
              p.kdrek,
              p.nmrek,
              p.keterangan || '-',
              p.usulan === 'Penambahan' ? formatAngkaIndo(p.jumlah) : '',
              p.usulan === 'Pengurangan' ? formatAngkaIndo(p.jumlah) : '',
              p.setuju || ''
            ]);
          });
        }

        // 2. BELANJA
        if (struct.belanjaSubgiat.length > 0) {
          hasDataInStruct = true;
          tableRows.push(['2', '2', 'BELANJA DAERAH', '', formatAngkaIndo(struct.totalPenambahan), formatAngkaIndo(struct.totalPengurangan), '']);

          struct.belanjaSubgiat.forEach((sg, idx) => {
            const sgPenambahan = sg.items.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
            const sgPengurangan = sg.items.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);

            tableRows.push([
              `2.${idx + 1}`,
              sg.kdsubgiat,
              sg.nmsubgiat,
              '',
              sgPenambahan ? formatAngkaIndo(sgPenambahan) : '',
              sgPengurangan ? formatAngkaIndo(sgPengurangan) : '',
              ''
            ]);

            sg.items.forEach(b => {
              tableRows.push([
                '',
                b.kdrek,
                b.nmrek,
                b.keterangan || '-',
                b.usulan === 'Penambahan' ? formatAngkaIndo(b.jumlah) : '',
                b.usulan === 'Pengurangan' ? formatAngkaIndo(b.jumlah) : '',
                b.setuju || ''
              ]);
            });
          });
        }

        // 3. PEMBIAYAAN
        if (struct.pembiayaan.length > 0) {
          hasDataInStruct = true;
          const totPenambahanPembiayaan = struct.pembiayaan.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
          const totPenguranganPembiayaan = struct.pembiayaan.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);
          
          tableRows.push(['3', '3', 'PEMBIAYAAN DAERAH', '', totPenambahanPembiayaan ? formatAngkaIndo(totPenambahanPembiayaan) : '', totPenguranganPembiayaan ? formatAngkaIndo(totPenguranganPembiayaan) : '', '']);

          struct.pembiayaan.forEach(pb => {
            tableRows.push([
              '',
              pb.kdrek,
              pb.nmrek,
              pb.keterangan || '-',
              pb.usulan === 'Penambahan' ? formatAngkaIndo(pb.jumlah) : '',
              pb.usulan === 'Pengurangan' ? formatAngkaIndo(pb.jumlah) : '',
              pb.setuju || ''
            ]);
          });
        }

        return hasDataInStruct;
      };

      if (selectedKdSkpd === DINKES_KODE) {
        // A. CETAK REKAPITULASI TOTAL SKPD DINAS KESEHATAN
        const totalSkpdStruct = buildReportStructure(filteredData);
        tableRows.push(['', '', 'REKAPITULASI SKPD DINAS KESEHATAN (TOTAL GABUNGAN)', '', '', '', '']);
        const hasTotalData = appendStructureToRows(totalSkpdStruct);
        if (!hasTotalData) {
          tableRows.push(['', '', '-- Tidak Ada Data Asistensi untuk SKPD --', '', '', '', '']);
        }

        // B. CETAK RINCIAN PER SUBUNIT DINAS KESEHATAN
        listSubunitDinkes.forEach((sub, sIdx) => {
          const kodeSub = sub.kd_subunit || sub.kdsubunit;
          const namaSub = sub.nm_subunit || sub.nmsubunit;
          const subItems = filteredData.filter(d => normalize(d.kdsubunit) === normalize(kodeSub));
          
          tableRows.push(['', '', '', '', '', '', '']); // Baris pemisah
          tableRows.push([`SUB ${sIdx + 1}`, kodeSub, `SUBUNIT: ${namaSub.toUpperCase()}`, '', '', '', '']);

          const struct = buildReportStructure(subItems);
          const hasSubData = appendStructureToRows(struct);
          if (!hasSubData) {
            tableRows.push(['', '', '-- Tidak Ada Data Asistensi --', '', '', '', '']);
          }
        });
      } else {
        const struct = buildReportStructure(filteredData);
        const hasData = appendStructureToRows(struct);
        if (!hasData) {
          tableRows.push(['', '', '-- Tidak Ada Data Asistensi --', '', '', '', '']);
        }
      }

      autoTable(doc, {
        startY: 26,
        head: [
          [
            { content: 'No', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'KODE', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'URAIAN', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'KETERANGAN ASISTENSI', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'PENAMBAHAN', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'PENGURANGAN', styles: { halign: 'center', valalign: 'middle' } },
            { content: 'STATUS', styles: { halign: 'center', valalign: 'middle' } }
          ]
        ],
        body: tableRows,
        theme: 'grid',
        styles: {
          fontSize: 7,
          cellPadding: 1.2,
          font: 'helvetica',
          textColor: [0, 0, 0],
          lineWidth: 0.1,
          lineColor: [120, 120, 120]
        },
        headStyles: {
          fillColor: [240, 240, 240],
          textColor: [0, 0, 0],
          fontStyle: 'bold'
        },
        columnStyles: {
          0: { halign: 'center', cellWidth: 8 },
          1: { cellWidth: 28 },
          2: { cellWidth: 'auto' },
          3: { cellWidth: 35 },
          4: { halign: 'right', cellWidth: 25 },
          5: { halign: 'right', cellWidth: 25 },
          6: { halign: 'center', cellWidth: 20 }
        },
        margin: { top: 26, left: 8, right: 8, bottom: 10 }
      });

      doc.save(`LAPORAN_HASIL_ASISTENSI_${selectedKdSkpd}_${new Date().getTime()}.pdf`);
    } catch (err) {
      console.error("Gagal download PDF:", err);
      alert("Terjadi kesalahan saat memproses PDF.");
    } finally {
      setDownloadingPdf(false);
    }
  };

  const handleExportExcel = () => {
    const wb = XLSX.utils.book_new();
    const rows = [];

    rows.push(['LAPORAN HASIL ASISTENSI APBD']);
    rows.push(['SKPD :', selectedSkpdObj?.nm_skpd || selectedSkpdObj?.nmskpd || '-']);
    rows.push(['STATUS :', selectedStatusObj?.nm_status || selectedStatusObj?.status || selectedStatusObj?.nmstatus || '-']);
    rows.push([]);
    rows.push(['No', 'KODE', 'URAIAN', 'KETERANGAN ASISTENSI', 'PENAMBAHAN', 'PENGURANGAN', 'STATUS']);

    const appendStructureToExcel = (struct) => {
      let hasData = false;

      if (struct.pendapatan.length > 0) {
        hasData = true;
        const totPenambahanPendapatan = struct.pendapatan.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
        const totPenguranganPendapatan = struct.pendapatan.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);
        rows.push(['1', '1', 'PENDAPATAN DAERAH', '', totPenambahanPendapatan || '', totPenguranganPendapatan || '', '']);
        struct.pendapatan.forEach(p => {
          rows.push(['', p.kdrek, p.nmrek, p.keterangan, p.usulan === 'Penambahan' ? p.jumlah : '', p.usulan === 'Pengurangan' ? p.jumlah : '', p.setuju]);
        });
      }

      if (struct.belanjaSubgiat.length > 0) {
        hasData = true;
        rows.push(['2', '2', 'BELANJA DAERAH', '', struct.totalPenambahan, struct.totalPengurangan, '']);
        struct.belanjaSubgiat.forEach((sg, idx) => {
          const sgPenambahan = sg.items.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
          const sgPengurangan = sg.items.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);
          rows.push([`2.${idx + 1}`, sg.kdsubgiat, sg.nmsubgiat, '', sgPenambahan || '', sgPengurangan || '', '']);
          sg.items.forEach(b => {
            rows.push(['', b.kdrek, b.nmrek, b.keterangan, b.usulan === 'Penambahan' ? b.jumlah : '', b.usulan === 'Pengurangan' ? b.jumlah : '', b.setuju]);
          });
        });
      }

      if (struct.pembiayaan.length > 0) {
        hasData = true;
        const totPenambahanPembiayaan = struct.pembiayaan.reduce((a, c) => a + (c.usulan === 'Penambahan' ? Number(c.jumlah || 0) : 0), 0);
        const totPenguranganPembiayaan = struct.pembiayaan.reduce((a, c) => a + (c.usulan === 'Pengurangan' ? Number(c.jumlah || 0) : 0), 0);
        rows.push(['3', '3', 'PEMBIAYAAN DAERAH', '', totPenambahanPembiayaan || '', totPenguranganPembiayaan || '', '']);
        struct.pembiayaan.forEach(pb => {
          rows.push(['', pb.kdrek, pb.nmrek, pb.keterangan, pb.usulan === 'Penambahan' ? pb.jumlah : '', pb.usulan === 'Pengurangan' ? pb.jumlah : '', pb.setuju]);
        });
      }

      return hasData;
    };

    if (selectedKdSkpd === DINKES_KODE) {
      // Rekap SKPD Dinas Kesehatan
      rows.push(['', '', 'REKAPITULASI SKPD DINAS KESEHATAN (TOTAL GABUNGAN)', '', '', '', '']);
      const totalSkpdStruct = buildReportStructure(filteredData);
      const hasTotalData = appendStructureToExcel(totalSkpdStruct);
      if (!hasTotalData) {
        rows.push(['', '', '-- Tidak Ada Data Asistensi untuk SKPD --', '', '', '', '']);
      }
      rows.push([]);

      // Per Subunit
      listSubunitDinkes.forEach((sub, sIdx) => {
        const kodeSub = sub.kd_subunit || sub.kdsubunit;
        const namaSub = sub.nm_subunit || sub.nmsubunit;
        const subItems = filteredData.filter(d => normalize(d.kdsubunit) === normalize(kodeSub));

        rows.push([`SUBUNIT ${sIdx + 1}`, kodeSub, namaSub.toUpperCase(), '', '', '', '']);
        const struct = buildReportStructure(subItems);
        const hasSubData = appendStructureToExcel(struct);
        if (!hasSubData) {
          rows.push(['', '', '-- Tidak Ada Data Asistensi --', '', '', '', '']);
        }
        rows.push([]);
      });
    } else {
      const struct = buildReportStructure(filteredData);
      const hasData = appendStructureToExcel(struct);
      if (!hasData) {
        rows.push(['', '', '-- Tidak Ada Data Asistensi --', '', '', '', '']);
      }
    }

    const ws = XLSX.utils.aoa_to_sheet(rows);
    XLSX.utils.book_append_sheet(wb, ws, 'Laporan Asistensi');
    XLSX.writeFile(wb, `Laporan_Asistensi_${selectedKdSkpd}.xlsx`);
  };

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between bg-slate-950/80 p-3.5 rounded-xl border border-slate-800">
        <div className="text-xs font-mono font-bold text-indigo-400 uppercase tracking-wider flex items-center gap-2">
          <FileText size={16} /> Page Frame Laporan Hasil Asistensi APBD
        </div>
        <div className="flex items-center gap-2">
          <button
            onClick={handleExportExcel}
            className="flex items-center gap-1.5 px-3.5 py-1.5 rounded-xl bg-emerald-950 hover:bg-emerald-900 text-emerald-300 border border-emerald-700 font-mono text-xs font-bold transition-all cursor-pointer"
          >
            <Download size={13} /> EXCEL
          </button>
          <button
            onClick={handleDownloadPDF}
            disabled={downloadingPdf}
            className="flex items-center gap-1.5 px-3.5 py-1.5 rounded-xl bg-cyan-950 hover:bg-cyan-900 text-cyan-300 border border-cyan-700 font-mono text-xs font-bold transition-all cursor-pointer disabled:opacity-50"
          >
            <Download size={13} /> {downloadingPdf ? 'GENERATING PDF...' : 'DOWNLOAD PDF'}
          </button>
        </div>
      </div>

      <div className="bg-white text-black p-6 rounded-xl font-sans text-xs shadow-2xl space-y-4 overflow-x-auto">
        <div className="text-center font-bold text-sm tracking-wide border-b-2 border-black pb-2 uppercase">
          LAPORAN HASIL ASISTENSI APBD
        </div>

        <div className="font-bold space-y-0.5 text-xs">
          <div className="flex">
            <span className="w-24">SKPD</span>
            <span>: {selectedSkpdObj?.nm_skpd || selectedSkpdObj?.nmskpd || '-'}</span>
          </div>
          <div className="flex">
            <span className="w-24">STATUS</span>
            <span>: {selectedStatusObj?.nm_status || selectedStatusObj?.status || selectedStatusObj?.nmstatus || '-'}</span>
          </div>
        </div>

        {selectedKdSkpd === DINKES_KODE ? (
          <div className="space-y-6">
            {/* 1. REKAPITULASI SKPD DINAS KESEHATAN */}
            <div className="space-y-2 border-2 border-indigo-600 p-2 rounded-lg bg-indigo-50/20">
              <div className="font-bold text-xs uppercase bg-indigo-100 p-1.5 border-b border-indigo-600 text-indigo-900">
                REKAPITULASI SKPD DINAS KESEHATAN (TOTAL GABUNGAN)
              </div>
              <TableReportView struct={buildReportStructure(filteredData)} />
            </div>

            {/* 2. PER SUBUNIT DINAS KESEHATAN */}
            {listSubunitDinkes.map((sub, sIdx) => {
              const kodeSub = sub.kd_subunit || sub.kdsubunit;
              const namaSub = sub.nm_subunit || sub.nmsubunit;
              const subItems = filteredData.filter(d => normalize(d.kdsubunit) === normalize(kodeSub));
              const struct = buildReportStructure(subItems);

              return (
                <div key={sIdx} className="space-y-2 border border-black p-2 rounded-lg">
                  <div className="font-bold text-xs uppercase bg-slate-100 p-1.5 border-b border-black">
                    SUBUNIT {sIdx + 1}: [{kodeSub}] {namaSub}
                  </div>
                  <TableReportView struct={struct} />
                </div>
              );
            })}
          </div>
        ) : (
          <TableReportView struct={buildReportStructure(filteredData)} />
        )}
      </div>
    </div>
  );
}

function TableReportView({ struct }) {
  const formatRupiah = (val) => {
    if (!val && val !== 0) return '';
    return Number(val).toLocaleString('id-ID', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  };

  const hasPendapatan = struct.pendapatan && struct.pendapatan.length > 0;
  const hasBelanja = struct.belanjaSubgiat && struct.belanjaSubgiat.length > 0;
  const hasPembiayaan = struct.pembiayaan && struct.pembiayaan.length > 0;

  if (!hasPendapatan && !hasBelanja && !hasPembiayaan) {
    return (
      <div className="p-4 text-center text-slate-500 font-mono text-xs italic bg-slate-50 border border-black">
        -- Tidak Ada Data Asistensi --
      </div>
    );
  }

  return (
    <table className="w-full border-collapse border border-black text-left text-[11px]">
      <thead>
        <tr className="bg-slate-100 border-b border-black text-center font-bold">
          <th className="border border-black p-1.5 w-8">No</th>
          <th className="border border-black p-1.5 w-36">KODE</th>
          <th className="border border-black p-1.5">URAIAN</th>
          <th className="border border-black p-1.5">KETERANGAN ASISTENSI</th>
          <th className="border border-black p-1.5 w-32 text-right">PENAMBAHAN</th>
          <th className="border border-black p-1.5 w-32 text-right">PENGURANGAN</th>
          <th className="border border-black p-1.5 w-24 text-center">STATUS</th>
        </tr>
      </thead>
      <tbody>
        {/* PENDAPATAN DAERAH */}
        {hasPendapatan && (
          <>
            <tr className="font-bold bg-slate-50">
              <td className="border border-black p-1 text-center">1</td>
              <td className="border border-black p-1">1</td>
              <td className="border border-black p-1">PENDAPATAN DAERAH</td>
              <td className="border border-black p-1"></td>
              <td className="border border-black p-1 text-right">
                {formatRupiah(struct.pendapatan.reduce((a,c)=>a+(c.usulan==='Penambahan'?Number(c.jumlah||0):0),0))}
              </td>
              <td className="border border-black p-1 text-right">
                {formatRupiah(struct.pendapatan.reduce((a,c)=>a+(c.usulan==='Pengurangan'?Number(c.jumlah||0):0),0))}
              </td>
              <td className="border border-black p-1"></td>
            </tr>
            {struct.pendapatan.map((item, idx) => (
              <tr key={idx}>
                <td className="border border-black p-1 text-center"></td>
                <td className="border border-black p-1 font-mono">{item.kdrek}</td>
                <td className="border border-black p-1">{item.nmrek}</td>
                <td className="border border-black p-1">{item.keterangan}</td>
                <td className="border border-black p-1 text-right">{item.usulan === 'Penambahan' ? formatRupiah(item.jumlah) : ''}</td>
                <td className="border border-black p-1 text-right">{item.usulan === 'Pengurangan' ? formatRupiah(item.jumlah) : ''}</td>
                <td className="border border-black p-1 text-center">{item.setuju}</td>
              </tr>
            ))}
          </>
        )}

        {/* BELANJA DAERAH */}
        {hasBelanja && (
          <>
            <tr className="font-bold bg-slate-50">
              <td className="border border-black p-1 text-center">2</td>
              <td className="border border-black p-1">2</td>
              <td className="border border-black p-1">BELANJA DAERAH</td>
              <td className="border border-black p-1"></td>
              <td className="border border-black p-1 text-right">{formatRupiah(struct.totalPenambahan)}</td>
              <td className="border border-black p-1 text-right">{formatRupiah(struct.totalPengurangan)}</td>
              <td className="border border-black p-1"></td>
            </tr>
            {struct.belanjaSubgiat.map((sg, sIdx) => (
              <React.Fragment key={sIdx}>
                <tr className="font-bold">
                  <td className="border border-black p-1 text-center">2.{sIdx + 1}</td>
                  <td className="border border-black p-1 font-mono">{sg.kdsubgiat}</td>
                  <td className="border border-black p-1">{sg.nmsubgiat}</td>
                  <td className="border border-black p-1"></td>
                  <td className="border border-black p-1 text-right">
                    {formatRupiah(sg.items.reduce((a,c)=>a+(c.usulan==='Penambahan'?Number(c.jumlah||0):0),0))}
                  </td>
                  <td className="border border-black p-1 text-right">
                    {formatRupiah(sg.items.reduce((a,c)=>a+(c.usulan==='Pengurangan'?Number(c.jumlah||0):0),0))}
                  </td>
                  <td className="border border-black p-1"></td>
                </tr>
                {sg.items.map((b, bIdx) => (
                  <tr key={bIdx}>
                    <td className="border border-black p-1 text-center"></td>
                    <td className="border border-black p-1 font-mono">{b.kdrek}</td>
                    <td className="border border-black p-1">{b.nmrek}</td>
                    <td className="border border-black p-1">{b.keterangan}</td>
                    <td className="border border-black p-1 text-right">{b.usulan === 'Penambahan' ? formatRupiah(b.jumlah) : ''}</td>
                    <td className="border border-black p-1 text-right">{b.usulan === 'Pengurangan' ? formatRupiah(b.jumlah) : ''}</td>
                    <td className="border border-black p-1 text-center">{b.setuju}</td>
                  </tr>
                ))}
              </React.Fragment>
            ))}
          </>
        )}

        {/* PEMBIAYAAN DAERAH */}
        {hasPembiayaan && (
          <>
            <tr className="font-bold bg-slate-50">
              <td className="border border-black p-1 text-center">3</td>
              <td className="border border-black p-1">3</td>
              <td className="border border-black p-1">PEMBIAYAAN DAERAH</td>
              <td className="border border-black p-1"></td>
              <td className="border border-black p-1 text-right">
                {formatRupiah(struct.pembiayaan.reduce((a,c)=>a+(c.usulan==='Penambahan'?Number(c.jumlah||0):0),0))}
              </td>
              <td className="border border-black p-1 text-right">
                {formatRupiah(struct.pembiayaan.reduce((a,c)=>a+(c.usulan==='Pengurangan'?Number(c.jumlah||0):0),0))}
              </td>
              <td className="border border-black p-1"></td>
            </tr>
            {struct.pembiayaan.map((pb, pIdx) => (
              <tr key={pIdx}>
                <td className="border border-black p-1 text-center"></td>
                <td className="border border-black p-1 font-mono">{pb.kdrek}</td>
                <td className="border border-black p-1">{pb.nmrek}</td>
                <td className="border border-black p-1">{pb.keterangan}</td>
                <td className="border border-black p-1 text-right">{pb.usulan === 'Penambahan' ? formatRupiah(pb.jumlah) : ''}</td>
                <td className="border border-black p-1 text-right">{pb.usulan === 'Pengurangan' ? formatRupiah(pb.jumlah) : ''}</td>
                <td className="border border-black p-1 text-center">{pb.setuju}</td>
              </tr>
            ))}
          </>
        )}
      </tbody>
    </table>
  );
}
