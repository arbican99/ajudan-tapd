import React, { useState, useEffect, useMemo } from 'react';
import { supabase } from '../supabaseClient';
import { 
  Layers, Calendar, RefreshCw, 
  TrendingDown, DollarSign, Building2,
  FileText, PieChart, CreditCard, CheckCircle2, Terminal, Filter
} from 'lucide-react';

export default function DashboardAPBD() {
  // --- STATE MANAGEMENT ---
  const [skpdList, setSkpdList] = useState([]);
  const [tahunList, setTahunList] = useState([]);
  const [statusList, setStatusList] = useState([]);
  
  const [selectedSkpd, setSelectedSkpd] = useState('REKAP'); 
  const [selectedTahun, setSelectedTahun] = useState('ALL');
  const [selectedStatus, setSelectedStatus] = useState('001');
  
  const [activeTab, setActiveTab] = useState(1); // 1: Pendapatan, 2: Belanja, 3: Pembiayaan
  const [loading, setLoading] = useState(false);
  const [rawApbdData, setRawApbdData] = useState([]);

  useEffect(() => {
    ambilDataFilter();
  }, []);

  useEffect(() => {
    fetchMainData();
  }, [selectedSkpd, selectedTahun, selectedStatus]);

  // Cek apakah status yang dipilih adalah mode APBD saja
  const isApbdOnly = useMemo(() => {
    const found = statusList.find(s => String(s.kd_status) === String(selectedStatus));
    return found ? String(found.status).trim().toUpperCase() === 'APBD' : false;
  }, [selectedStatus, statusList]);

  // --- 1. AMBIL DATA FILTER (SKPD, TAHUN, & STATUS) ---
  const ambilDataFilter = async () => {
    try {
      // Fetch tblstatus
      const { data: dataStatus, error: errStatus } = await supabase
        .from('tblstatus')
        .select('kd_status, status')
        .order('kd_status', { ascending: true });

      if (errStatus) console.error("Gagal load tblstatus:", errStatus.message);
      if (dataStatus && dataStatus.length > 0) {
        setStatusList(dataStatus);
        setSelectedStatus(dataStatus[0].kd_status);
      }

      // Fetch SKPD & Tahun secara bertahap
      let semuaData = [];
      let page = 0;
      const tumpukanPerQuery = 1000;
      let masihAdaData = true;

      while (masihAdaData) {
        const dariBaris = page * tumpukanPerQuery;
        const sampaiBaris = dariBaris + tumpukanPerQuery - 1;

        const { data, error } = await supabase
          .from('apbd')
          .select('kdskpd, nmskpd, tahun')
          .range(dariBaris, sampaiBaris);

        if (error) throw error;

        if (data && data.length > 0) {
          semuaData = [...semuaData, ...data];
          if (data.length < tumpukanPerQuery) {
            masihAdaData = false;
          } else {
            page++;
          }
        } else {
          masihAdaData = false;
        }
      }

      if (semuaData.length > 0) {
        const uniqueSkpd = [];
        const mapSkpd = new Map();
        const uniqueTahun = new Set();

        semuaData.forEach(item => {
          const kodeClean = item.kdskpd ? String(item.kdskpd).trim() : '';

          if (kodeClean && !mapSkpd.has(kodeClean)) {
            mapSkpd.set(kodeClean, true);
            uniqueSkpd.push({
              kdskpd: kodeClean,
              nmskpd: item.nmskpd || 'Tanpa Nama SKPD'
            });
          }
          if (item.tahun) uniqueTahun.add(String(item.tahun));
        });

        uniqueSkpd.sort((a, b) => (a.kdskpd || '').localeCompare(b.kdskpd || ''));
        setSkpdList(uniqueSkpd);

        const sortedTahun = Array.from(uniqueTahun).sort();
        setTahunList(sortedTahun);

        if (sortedTahun.length > 0) {
          setSelectedTahun(sortedTahun[sortedTahun.length - 1]);
        }
      }
    } catch (err) {
      console.error("Gagal mengambil filter:", err.message);
    }
  };

  // --- 2. FETCH DATA UTAMA DARI SUPABASE ---
  const fetchMainData = async () => {
    setLoading(true);
    try {
      let semuaData = [];
      let page = 0;
      const tumpukanPerQuery = 1000;
      let masihAdaData = true;

      while (masihAdaData) {
        const dariBaris = page * tumpukanPerQuery;
        const sampaiBaris = dariBaris + tumpukanPerQuery - 1;

        let query = supabase.from('apbd').select('*');

        if (selectedSkpd !== 'REKAP') {
          query = query.eq('kdskpd', selectedSkpd);
        }
        if (selectedTahun !== 'ALL') {
          query = query.eq('tahun', String(selectedTahun));
        }

        const { data, error } = await query.range(dariBaris, sampaiBaris);

        if (error) throw error;

        if (data && data.length > 0) {
          semuaData = [...semuaData, ...data];
          if (data.length < tumpukanPerQuery) {
            masihAdaData = false;
          } else {
            page++;
          }
        } else {
          masihAdaData = false;
        }
      }

      setRawApbdData(semuaData);
    } catch (err) {
      console.error("Gagal memuat data APBD:", err.message);
    } finally {
      setLoading(false);
    }
  };

  // --- HELPER PARSE ANGKA REAL ---
  const parseAngkaReal = (val) => {
    if (val === null || val === undefined) return 0;
    if (typeof val === 'number') return val;
    const cleaned = String(val).replace(/[^0-9.-]/g, '');
    const parsed = parseFloat(cleaned);
    return isNaN(parsed) ? 0 : parsed;
  };

  // --- HELPER FORMAT CURRENCY ---
  const formatRupiah = (val) => {
    const num = parseAngkaReal(val);
    return new Intl.NumberFormat('id-ID', {
      style: 'currency',
      currency: 'IDR',
      maximumFractionDigits: 0
    }).format(num);
  };

  const formatSingkat = (val) => {
    const num = parseAngkaReal(val);
    const absVal = Math.abs(num);
    const prefix = num < 0 ? '-Rp ' : 'Rp ';
    if (absVal >= 1e12) return `${prefix}${(absVal / 1e12).toFixed(2)} T`;
    if (absVal >= 1e9) return `${prefix}${(absVal / 1e9).toFixed(2)} M`;
    if (absVal >= 1e6) return `${prefix}${(absVal / 1e6).toFixed(2)} Jt`;
    return formatRupiah(num);
  };

  // --- AKUMULASI DATA DASHBOARD ---
  const summaryData = useMemo(() => {
    let totalAwalBelanja = 0;
    let totalAkhirBelanja = 0;

    let totalAwalPendapatan = 0;
    let totalAkhirPendapatan = 0;

    let totalAwalPenerimaanPembiayaan = 0;
    let totalAkhirPenerimaanPembiayaan = 0;
    let totalAwalPengeluaranPembiayaan = 0;
    let totalAkhirPengeluaranPembiayaan = 0;

    const belanjaKategori = {
      operasi: { awal: 0, akhir: 0 },
      modal: { awal: 0, akhir: 0 },
      tidakTerduga: { awal: 0, akhir: 0 },
      transfer: { awal: 0, akhir: 0 }
    };

    const pendapatanRincian = {
      pad: { awal: 0, akhir: 0 },
      dau: { awal: 0, akhir: 0 },
      dak: { awal: 0, akhir: 0 },
      dbh: { awal: 0, akhir: 0 },
      lain: { awal: 0, akhir: 0 }
    };

    const skpdTableMap = new Map();

    rawApbdData.forEach(item => {
      const awal = parseAngkaReal(item.awal || item.apbd_awal || item.anggaran_awal);
      const akhirRaw = parseAngkaReal(item.akhir || item.apbd_akhir || item.anggaran_akhir);
      const akhir = isApbdOnly ? awal : akhirRaw;
      
      const realisasi = parseAngkaReal(
        item.real ?? 
        item.realisasi ?? 
        item.nilai_realisasi ?? 
        0
      );

      const kdrek = item.kdrek ? String(item.kdrek).trim() : '';
      const nmrek = item.nmrek ? String(item.nmrek).toLowerCase() : '';
      const kdskpd = item.kdskpd ? String(item.kdskpd).trim() : 'LAINNYA';
      const nmskpd = item.nmskpd || 'SKPD Tidak Teridentifikasi';

      if (!skpdTableMap.has(kdskpd)) {
        skpdTableMap.set(kdskpd, {
          kdskpd,
          nmskpd,
          pendapatan: { awal: 0, akhir: 0, realisasi: 0 },
          belanja: { awal: 0, akhir: 0, realisasi: 0 },
          pembiayaan: { awal: 0, akhir: 0, realisasi: 0 }
        });
      }
      const skpdObj = skpdTableMap.get(kdskpd);

      // 1. REKENING BELANJA (KODE 5)
      if (kdrek.startsWith('5') || nmrek.includes('belanja')) {
        totalAwalBelanja += awal;
        totalAkhirBelanja += akhir;

        skpdObj.belanja.awal += awal;
        skpdObj.belanja.akhir += akhir;
        skpdObj.belanja.realisasi += realisasi;

        if (kdrek.startsWith('5.1') || nmrek.includes('operasi')) {
          belanjaKategori.operasi.awal += awal;
          belanjaKategori.operasi.akhir += akhir;
        } else if (kdrek.startsWith('5.2') || nmrek.includes('modal')) {
          belanjaKategori.modal.awal += awal;
          belanjaKategori.modal.akhir += akhir;
        } else if (kdrek.startsWith('5.3') || nmrek.includes('tidak terduga')) {
          belanjaKategori.tidakTerduga.awal += awal;
          belanjaKategori.tidakTerduga.akhir += akhir;
        } else if (kdrek.startsWith('5.4') || nmrek.includes('transfer')) {
          belanjaKategori.transfer.awal += awal;
          belanjaKategori.transfer.akhir += akhir;
        } else {
          belanjaKategori.operasi.awal += awal;
          belanjaKategori.operasi.akhir += akhir;
        }
      } 
      // 2. REKENING PENDAPATAN (KODE 4)
      else if (kdrek.startsWith('4') || nmrek.includes('pendapatan')) {
        totalAwalPendapatan += awal;
        totalAkhirPendapatan += akhir;

        skpdObj.pendapatan.awal += awal;
        skpdObj.pendapatan.akhir += akhir;
        skpdObj.pendapatan.realisasi += realisasi;

        if (kdrek.startsWith('4.1') || nmrek.includes('pad') || nmrek.includes('asli daerah')) {
          pendapatanRincian.pad.awal += awal;
          pendapatanRincian.pad.akhir += akhir;
        } else if (nmrek.includes('dau') || nmrek.includes('alokasi umum')) {
          pendapatanRincian.dau.awal += awal;
          pendapatanRincian.dau.akhir += akhir;
        } else if (nmrek.includes('dak') || nmrek.includes('alokasi khusus')) {
          pendapatanRincian.dak.awal += awal;
          pendapatanRincian.dak.akhir += akhir;
        } else if (nmrek.includes('dbh') || nmrek.includes('bagi hasil')) {
          pendapatanRincian.dbh.awal += awal;
          pendapatanRincian.dbh.akhir += akhir;
        } else {
          pendapatanRincian.lain.awal += awal;
          pendapatanRincian.lain.akhir += akhir;
        }
      } 
      // 3. REKENING PEMBIAYAAN (KODE 6)
      else if (kdrek.startsWith('6') || nmrek.includes('pembiayaan')) {
        skpdObj.pembiayaan.awal += awal;
        skpdObj.pembiayaan.akhir += akhir;
        skpdObj.pembiayaan.realisasi += realisasi;

        if (kdrek.startsWith('6.1') || nmrek.includes('penerimaan')) {
          totalAwalPenerimaanPembiayaan += awal;
          totalAkhirPenerimaanPembiayaan += akhir;
        } else if (kdrek.startsWith('6.2') || nmrek.includes('pengeluaran')) {
          totalAwalPengeluaranPembiayaan += awal;
          totalAkhirPengeluaranPembiayaan += akhir;
        } else {
          totalAwalPenerimaanPembiayaan += awal;
          totalAkhirPenerimaanPembiayaan += akhir;
        }
      }
    });

    const defisitAwal = totalAwalPendapatan - totalAwalBelanja;
    const defisitAkhir = totalAkhirPendapatan - totalAkhirBelanja;

    const pembiayaanBersihAwal = totalAwalPenerimaanPembiayaan - totalAwalPengeluaranPembiayaan;
    const pembiayaanBersihAkhir = totalAkhirPenerimaanPembiayaan - totalAkhirPengeluaranPembiayaan;

    const skpdTableArray = Array.from(skpdTableMap.values()).sort((a, b) => a.kdskpd.localeCompare(b.kdskpd));

    return {
      totalAwalBelanja,
      totalAkhirBelanja,
      totalAwalPendapatan,
      totalAkhirPendapatan,
      defisitAwal,
      defisitAkhir,
      pembiayaanBersihAwal,
      pembiayaanBersihAkhir,
      belanjaKategori,
      pendapatanRincian,
      skpdTableArray
    };
  }, [rawApbdData, isApbdOnly]);

  const currentSkpdName = useMemo(() => {
    if (selectedSkpd === 'REKAP') return 'REKAPITULASI SELURUH SKPD';
    const found = skpdList.find(s => s.kdskpd === selectedSkpd);
    return found ? `${found.kdskpd} - ${found.nmskpd}` : selectedSkpd;
  }, [selectedSkpd, skpdList]);

  const currentStatusName = useMemo(() => {
    const found = statusList.find(s => String(s.kd_status) === String(selectedStatus));
    return found ? found.status : 'APBD';
  }, [selectedStatus, statusList]);

  return (
    <div className="min-h-screen bg-[#030712] text-slate-100 font-mono p-4 md:p-6 select-none">
      
      {/* HEADER DASHBOARD */}
      <header className="mb-5 border-b border-cyan-500/30 pb-3 flex flex-col md:flex-row md:items-center md:justify-between gap-3">
        <div>
          <div className="flex items-center gap-2">
            <Layers className="w-7 h-7 text-cyan-400 animate-pulse" />
            <h1 className="text-lg md:text-xl font-extrabold tracking-wider bg-clip-text text-transparent bg-gradient-to-r from-cyan-300 via-amber-300 to-amber-500 uppercase">
              REKAP BELANJA DAERAH (Regional Expenditure Recap)
            </h1>
          </div>
          <p className="text-xs text-cyan-300/80 mt-1 flex items-center gap-1.5">
            <Building2 className="w-3.5 h-3.5 text-cyan-400" /> Mode Tampilan: <span className="text-amber-300 font-bold">{currentSkpdName}</span>
            <span className="text-cyan-600">|</span> Tahapan: <span className="text-emerald-400 font-bold">{currentStatusName}</span>
          </p>
        </div>

        <button
          onClick={fetchMainData}
          disabled={loading}
          className="self-start md:self-auto px-3 py-2 bg-cyan-950/80 hover:bg-cyan-900 border border-cyan-500/40 rounded-lg text-cyan-200 transition-all cursor-pointer flex items-center gap-1.5 text-xs shadow-[0_0_10px_rgba(34,211,238,0.2)]"
        >
          <RefreshCw className={`w-3.5 h-3.5 text-cyan-300 ${loading ? 'animate-spin' : ''}`} />
          <span className="font-bold">Sync Data</span>
        </button>
      </header>

      {/* FILTER BOX NETRAL */}
      <div className="mb-6 bg-[#0a0f1d] border border-slate-800 rounded-xl p-3 md:p-4 shadow-lg flex flex-col md:flex-row items-center justify-between gap-3 md:gap-4">
        
        {/* FILTER SKPD */}
        <div className="w-full md:flex-1 space-y-1">
          <label className="text-[10px] md:text-xs text-cyan-400 font-bold uppercase tracking-wider flex items-center gap-1">
            <Terminal className="w-3.5 h-3.5 text-cyan-400" /> FILTER SKPD
          </label>
          <div className="relative w-full">
            <select
              value={selectedSkpd}
              onChange={(e) => setSelectedSkpd(e.target.value)}
              className="w-full bg-[#030712] text-cyan-100 border border-slate-700 rounded-lg px-4 py-2 text-xs md:text-sm font-bold focus:outline-none focus:border-cyan-400 cursor-pointer appearance-none truncate shadow-inner"
            >
              <option value="REKAP">» REKAPITULASI BELANJA DAERAH</option>
              {skpdList.map((skpd) => (
                <option key={skpd.kdskpd} value={skpd.kdskpd}>
                  » {skpd.kdskpd} - {skpd.nmskpd}
                </option>
              ))}
            </select>
          </div>
        </div>

        {/* FILTER TAHANAN APBD (tblstatus) */}
        <div className="w-full md:w-56 space-y-1">
          <label className="text-[10px] md:text-xs text-emerald-400 font-bold uppercase tracking-wider flex items-center gap-1">
            <Filter className="w-3.5 h-3.5 text-emerald-400" /> TAHAPAN APBD
          </label>
          <div className="relative w-full">
            <select
              value={selectedStatus}
              onChange={(e) => setSelectedStatus(e.target.value)}
              className="w-full bg-[#030712] text-emerald-300 border border-slate-700 rounded-lg px-3 py-2 text-xs md:text-sm font-bold focus:outline-none focus:border-emerald-400 cursor-pointer appearance-none shadow-inner"
            >
              {statusList.map((st) => (
                <option key={st.kd_status} value={st.kd_status}>
                  » {st.status}
                </option>
              ))}
            </select>
          </div>
        </div>

        {/* FILTER TAHUN */}
        <div className="w-full md:w-48 space-y-1">
          <label className="text-[10px] md:text-xs text-cyan-400 font-bold uppercase tracking-wider flex items-center gap-1">
            <Calendar className="w-3.5 h-3.5 text-cyan-400" /> TAHUN ANGGARAN
          </label>
          <div className="relative w-full">
            <select
              value={selectedTahun}
              onChange={(e) => setSelectedTahun(e.target.value)}
              className="w-full bg-[#030712] text-cyan-100 border border-slate-700 rounded-lg px-3 py-2 text-xs md:text-sm font-bold focus:outline-none focus:border-cyan-400 cursor-pointer appearance-none shadow-inner"
            >
              <option value="ALL">» SEMUA TAHUN</option>
              {tahunList.map((th) => (
                <option key={th} value={th}>
                  » TAHUN {th}
                </option>
              ))}
            </select>
          </div>
        </div>

      </div>

      {/* MAIN CONTENT AREA */}
      {loading ? (
        <div className="h-80 flex flex-col items-center justify-center gap-3">
          <div className="w-10 h-10 border-4 border-amber-400 border-t-transparent rounded-full animate-spin"></div>
          <p className="text-amber-300 text-xs font-bold animate-pulse">Processing APBD Data...</p>
        </div>
      ) : (
        <div className="space-y-4">
          
          {/* BARIS 1: 5 CARDS RING CHART (BOX NETRAL DARK SLATE) */}
          <div className="grid grid-cols-1 md:grid-cols-3 lg:grid-cols-5 gap-3.5">
            
            {/* CARD 1: TOTAL BELANJA */}
            <div className="bg-[#0a0f1d] border border-slate-800 rounded-xl p-3 flex flex-col justify-between shadow-md hover:border-amber-500/50 transition-colors">
              <div className="text-center">
                <h3 className="text-[11px] font-black text-amber-400 uppercase tracking-wider">TOTAL BELANJA</h3>
                <div className="text-[9px] text-slate-400 flex justify-between px-1 mt-0.5 font-bold">
                  {isApbdOnly ? <span>Nilai APBD</span> : <><span>APBD Awal</span><span>APBD Akhir</span></>}
                </div>
              </div>

              <div className="my-2.5 flex justify-center items-center">
                <div className="w-20 h-20 rounded-full border-4 border-slate-900 border-t-amber-400 border-r-amber-500 flex flex-col items-center justify-center shadow-[0_0_15px_rgba(251,191,36,0.5)] bg-[#030712]">
                  <span className="text-sm font-black text-amber-300">
                    {isApbdOnly ? '100%' : `${summaryData.totalAwalBelanja > 0 ? ((summaryData.totalAkhirBelanja / summaryData.totalAwalBelanja) * 100).toFixed(0) : 100}%`}
                  </span>
                  <span className="text-[8px] font-extrabold text-amber-400 uppercase">{isApbdOnly ? 'APBD' : 'Progres'}</span>
                </div>
              </div>

              <div className="space-y-1 text-[11px] border-t border-slate-800/80 pt-2">
                <div className="flex justify-between">
                  <span className="text-slate-400 font-bold">Awal:</span>
                  <span className="text-slate-200 font-extrabold">{formatSingkat(summaryData.totalAwalBelanja)}</span>
                </div>
                {!isApbdOnly && (
                  <div className="flex justify-between">
                    <span className="text-slate-400 font-bold">Akhir:</span>
                    <span className="font-black text-amber-400">{formatSingkat(summaryData.totalAkhirBelanja)}</span>
                  </div>
                )}
              </div>
            </div>

            {/* CARD 2: BELANJA OPERASI */}
            <div className="bg-[#0a0f1d] border border-slate-800 rounded-xl p-3 flex flex-col justify-between shadow-md hover:border-orange-500/50 transition-colors">
              <div className="text-center">
                <h3 className="text-[11px] font-black text-orange-400 uppercase tracking-wider">BELANJA OPERASI</h3>
                <div className="text-[9px] text-slate-400 flex justify-between px-1 mt-0.5 font-bold">
                  {isApbdOnly ? <span>Nilai APBD</span> : <><span>APBD Awal</span><span>APBD Akhir</span></>}
                </div>
              </div>

              <div className="my-2.5 flex justify-center items-center">
                <div className="w-20 h-20 rounded-full border-4 border-slate-900 border-t-orange-400 border-l-orange-500 flex items-center justify-center shadow-[0_0_15px_rgba(251,146,60,0.5)] bg-[#030712]">
                  <span className="text-xs font-black text-orange-300">Operasi</span>
                </div>
              </div>

              <div className="space-y-1 text-[11px] border-t border-slate-800/80 pt-2">
                <div className="flex justify-between">
                  <span className="text-slate-400 font-bold">Awal:</span>
                  <span className="text-slate-200 font-extrabold">{formatSingkat(summaryData.belanjaKategori.operasi.awal)}</span>
                </div>
                {!isApbdOnly && (
                  <div className="flex justify-between">
                    <span className="text-slate-400 font-bold">Akhir:</span>
                    <span className="font-black text-orange-400">{formatSingkat(summaryData.belanjaKategori.operasi.akhir)}</span>
                  </div>
                )}
              </div>
            </div>

            {/* CARD 3: BELANJA MODAL */}
            <div className="bg-[#0a0f1d] border border-slate-800 rounded-xl p-3 flex flex-col justify-between shadow-md hover:border-cyan-500/50 transition-colors">
              <div className="text-center">
                <h3 className="text-[11px] font-black text-cyan-400 uppercase tracking-wider">BELANJA MODAL</h3>
                <div className="text-[9px] text-slate-400 flex justify-between px-1 mt-0.5 font-bold">
                  {isApbdOnly ? <span>Nilai APBD</span> : <><span>APBD Awal</span><span>APBD Akhir</span></>}
                </div>
              </div>

              <div className="my-2.5 flex justify-center items-center">
                <div className="w-20 h-20 rounded-full border-4 border-slate-900 border-t-cyan-400 border-b-cyan-500 flex items-center justify-center shadow-[0_0_15px_rgba(34,211,238,0.5)] bg-[#030712]">
                  <span className="text-xs font-black text-cyan-300">Modal</span>
                </div>
              </div>

              <div className="space-y-1 text-[11px] border-t border-slate-800/80 pt-2">
                <div className="flex justify-between">
                  <span className="text-slate-400 font-bold">Awal:</span>
                  <span className="text-slate-200 font-extrabold">{formatSingkat(summaryData.belanjaKategori.modal.awal)}</span>
                </div>
                {!isApbdOnly && (
                  <div className="flex justify-between">

            <div className="border-b border-slate-800 pb-2 flex items-center justify-between">
              <h2 className="text-xs md:text-sm font-black text-emerald-400 uppercase tracking-wider flex items-center gap-2">
                <CheckCircle2 className="w-5 h-5 text-emerald-400 animate-pulse" />
                PEMBIAYAAN BERSIH (PENERIMAAN - PENGELUARAN)
              </h2>
              <span className="text-[10px] text-slate-400 font-bold">
                {isApbdOnly ? 'Tampilan Data APBD' : 'Awal (Initial) vs Akhir (Revised)'}
              </span>
            </div>

            <div className="space-y-1">
              <div className="flex justify-between text-xs font-extrabold text-slate-200">
                <span className="text-emerald-400">REKAP PEMBIAYAAN BERSIH</span>
                <span className="text-emerald-300">
                  {isApbdOnly 
                    ? `Jumlah: ${formatSingkat(summaryData.pembiayaanBersihAwal)}` 
                    : `Awal: ${formatSingkat(summaryData.pembiayaanBersihAwal)} / Akhir: ${formatSingkat(summaryData.pembiayaanBersihAkhir)}`}
                </span>
              </div>
              <div className="w-full h-6 bg-[#030712] rounded-lg overflow-hidden border border-slate-800 flex p-0.5 shadow-inner">
                <div 
                  className="h-full bg-gradient-to-r from-emerald-600 via-emerald-400 to-teal-300 rounded transition-all duration-500 flex items-center justify-end pr-2 text-[10px] font-black text-slate-950 shadow-[0_0_18px_rgba(52,211,153,0.9)]"
                  style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pembiayaanBersihAwal !== 0 ? (summaryData.pembiayaanBersihAkhir / summaryData.pembiayaanBersihAwal) * 100 : 100))}%` }}
                >
                  {formatSingkat(isApbdOnly ? summaryData.pembiayaanBersihAwal : summaryData.pembiayaanBersihAkhir)}
                </div>
              </div>
            </div>
          </div>

          {/* BARIS 4: PENDAPATAN DAERAH (BOX NETRAL, BATANG NEON AMBER/YELLOW DOMINAN) */}

            <div className="border-b border-slate-800 pb-2 flex items-center justify-between">
              <h2 className="text-xs md:text-sm font-black text-amber-400 uppercase tracking-wider flex items-center gap-2">
                <DollarSign className="w-5 h-5 text-amber-400" />
                PENDAPATAN DAERAH (REGIONAL REVENUE)
              </h2>
              <span className="text-[10px] text-slate-400 font-bold">
                {isApbdOnly ? 'Tampilan Data APBD' : 'Awal (Initial) vs Akhir (Revised)'}
              </span>
            </div>

            <div className="space-y-3">
              
              {/* 0. REKAP PENDAPATAN DAERAH */}

                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.totalAwalPendapatan > 0 ? (summaryData.totalAkhirPendapatan / summaryData.totalAwalPendapatan) * 100 : 100))}%` }}

                <div className="flex justify-between text-[11px] text-amber-200 font-bold">
                  <span>1. PENDAPATAN ASLI DAERAH (PAD)</span>
                  <span>{isApbdOnly ? formatSingkat(summaryData.pendapatanRincian.pad.awal) : `Awal: ${formatSingkat(summaryData.pendapatanRincian.pad.awal)} / Akhir: ${formatSingkat(summaryData.pendapatanRincian.pad.akhir)}`}</span>
                </div>
                <div className="w-full h-5 bg-[#030712] rounded-lg overflow-hidden border border-slate-800 flex p-0.5 shadow-inner">
                  <div 
                    className="h-full bg-gradient-to-r from-amber-600 to-amber-400 rounded transition-all duration-500 shadow-[0_0_10px_rgba(245,158,11,0.6)]"
                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pendapatanRincian.pad.awal > 0 ? (summaryData.pendapatanRincian.pad.akhir / summaryData.pendapatanRincian.pad.awal) * 100 : 80))}%` }}
                  ></div>
                </div>
              </div>

              <div className="space-y-1 pl-4">
                <div className="flex justify-between text-[11px] text-amber-200/90 font-bold">
                  <span>2. DANA ALOKASI UMUM (DAU)</span>
                  <span>{isApbdOnly ? formatSingkat(summaryData.pendapatanRincian.dau.awal) : `Awal: ${formatSingkat(summaryData.pendapatanRincian.dau.awal)} / Akhir: ${formatSingkat(summaryData.pendapatanRincian.dau.akhir)}`}</span>
                </div>
                <div className="w-full h-4 bg-[#030712] rounded-lg overflow-hidden border border-slate-800 flex p-0.5">
                  <div 
                    className="h-full bg-gradient-to-r from-yellow-600 to-yellow-400 rounded transition-all duration-500 shadow-[0_0_8px_rgba(234,179,8,0.5)]"
                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pendapatanRincian.dau.awal > 0 ? (summaryData.pendapatanRincian.dau.akhir / summaryData.pendapatanRincian.dau.awal) * 100 : 85))}%` }}
                  ></div>
                </div>
              </div>

                    className="h-full bg-amber-500/80 rounded transition-all duration-500"
                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pendapatanRincian.dak.awal > 0 ? (summaryData.pendapatanRincian.dak.akhir / summaryData.pendapatanRincian.dak.awal) * 100 : 60))}%` }}
                  ></div>

                <div className="flex justify-between text-[11px] text-amber-200/80 font-bold">
                  <span>4. DANA BAGI HASIL (DBH)</span>
                  <span>{isApbdOnly ? formatSingkat(summaryData.pendapatanRincian.dbh.awal) : `Awal: ${formatSingkat(summaryData.pendapatanRincian.dbh.awal)} / Akhir: ${formatSingkat(summaryData.pendapatanRincian.dbh.akhir)}`}</span>

                    className="h-full bg-amber-600/70 rounded transition-all duration-500"
                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pendapatanRincian.dbh.awal > 0 ? (summaryData.pendapatanRincian.dbh.akhir / summaryData.pendapatanRincian.dbh.awal) * 100 : 50))}%` }}
                  ></div>

                  <span>5. LAIN-LAIN PENDAPATAN SAH</span>
                  <span>{isApbdOnly ? formatSingkat(summaryData.pendapatanRincian.lain.awal) : `Awal: ${formatSingkat(summaryData.pendapatanRincian.lain.awal)} / Akhir: ${formatSingkat(summaryData.pendapatanRincian.lain.akhir)}`}</span>

                  <div 
                    className="h-full bg-amber-700/70 rounded transition-all duration-500"
                    style={{ width: `${Math.min(100, isApbdOnly ? 100 : (summaryData.pendapatanRincian.lain.awal > 0 ? (summaryData.pendapatanRincian.lain.akhir / summaryData.pendapatanRincian.lain.awal) * 100 : 40))}%` }}
                  ></div>

            </div>
          </div>

          {/* BARIS 5: FRAME TABEL PER SKPD (BOX NETRAL) */}
          <div className="bg-[#0a0f1d] border border-slate-800 rounded-xl p-4 shadow-md space-y-4">
            
            {/* TAB BUTTONS */}
            <div className="flex flex-wrap items-center gap-2 border-b border-slate-800 pb-3">
              <button
                onClick={() => setActiveTab(1)}
                className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-bold transition-all cursor-pointer border ${
                  activeTab === 1 
                    ? 'bg-amber-500/20 text-amber-300 border-amber-500/60 shadow-[0_0_10px_rgba(245,158,11,0.3)]'

                }`}

                <FileText className="w-4 h-4" />
                <span>PAGE 1: PENDAPATAN DAERAH</span>
              </button>

              <button
                onClick={() => setActiveTab(2)}
                className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-bold transition-all cursor-pointer border ${
                  activeTab === 2 
                    ? 'bg-cyan-500/20 text-cyan-300 border-cyan-500/60 shadow-[0_0_10px_rgba(6,182,212,0.3)]' 
                    : 'bg-[#030712] text-slate-400 border-slate-800 hover:text-white'
                }`}
              >
                <PieChart className="w-4 h-4" />
                <span>PAGE 2: BELANJA DAERAH</span>
              </button>

              <button
                onClick={() => setActiveTab(3)}
                className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-bold transition-all cursor-pointer border ${
                  activeTab === 3 
                    ? 'bg-emerald-500/20 text-emerald-300 border-emerald-500/60 shadow-[0_0_10px_rgba(16,185,129,0.3)]' 
                    : 'bg-[#030712] text-slate-400 border-slate-800 hover:text-white'
                }`}
              >
                <CreditCard className="w-4 h-4" />
                <span>PAGE 3: PEMBIAYAAN DAERAH</span>
              </button>
            </div>

            {/* TABEL REALISASI PER SKPD */}
            <div className="overflow-x-auto rounded-lg border border-slate-800 shadow-md">
              <table className="w-full text-left text-xs text-slate-200 border-collapse">
                <thead>
                  <tr className="bg-[#030712] border-b border-slate-800 text-slate-300 uppercase text-[10px] tracking-wider">
                    <th className="p-3 font-extrabold text-center border-r border-slate-800">SKPD / UNIT KERJA</th>
                    <th className="p-3 font-extrabold text-center border-r border-slate-800">
                      {isApbdOnly ? 'ANGGARAN APBD' : 'APBD AWAL'}
                    </th>
                    {!isApbdOnly && (
                      <>
                        <th className="p-3 font-extrabold text-center border-r border-slate-800">APBD AKHIR</th>

                      </>
                    )}
                    <th className="p-3 font-extrabold text-center border-r border-slate-800">REALISASI</th>
                    <th className="p-3 font-extrabold text-center">SISA ANGGARAN</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-800/60 bg-[#070b16]">
                  {summaryData.skpdTableArray.length === 0 ? (
                    <tr>
                      <td colSpan={isApbdOnly ? "4" : "6"} className="p-4 text-center text-slate-400 italic font-semibold">
                        Tidak ada data rincian SKPD untuk ditampilkan.
                      </td>
                    </tr>
                  ) : (
                    summaryData.skpdTableArray.map((row) => {
                      const dataKategori = activeTab === 1 
                        ? row.pendapatan 
                        : activeTab === 2 
                          ? row.belanja 
                          : row.pembiayaan;

                      const berlebihBerkurang = dataKategori.akhir - dataKategori.awal;
                      const sisaAnggaran = (isApbdOnly ? dataKategori.awal : dataKategori.akhir) - dataKategori.realisasi;

                      return (
                        <tr key={row.kdskpd} className="hover:bg-slate-800/40 transition-colors">
                          <td className="p-2.5 font-bold text-slate-100 border-r border-slate-800/80">
                            <div className="text-xs">{row.kdskpd}</div>
                            <div className="text-[10px] text-slate-400 truncate max-w-xs">{row.nmskpd}</div>

                          <td className="p-2.5 text-right font-bold text-slate-200 border-r border-slate-800/80">
                            {formatRupiah(dataKategori.awal)}
                          </td>
                          {!isApbdOnly && (
                            <>
                              <td className="p-2.5 text-right font-bold text-white border-r border-slate-800/80">
                                {formatRupiah(dataKategori.akhir)}
                              </td>
                              <td className={`p-2.5 text-right font-black border-r border-slate-800/80 ${berlebihBerkurang >= 0 ? 'text-emerald-400' : 'text-rose-400'}`}>
                                {berlebihBerkurang >= 0 ? `+${formatRupiah(berlebihBerkurang)}` : formatRupiah(berlebihBerkurang)}
                              </td>
                            </>
                          )}
                          <td className="p-2.5 text-right font-black text-amber-300 border-r border-slate-800/80 bg-amber-500/5">
                            {formatRupiah(dataKategori.realisasi)}
                          </td>
                          <td className={`p-2.5 text-right font-black ${sisaAnggaran >= 0 ? 'text-cyan-300' : 'text-amber-400'}`}>
                            {formatRupiah(sisaAnggaran)}
                          </td>
                        </tr>
                      );
                    })
                  )}
                </tbody>
              </table>
            </div>

          </div>

        </div>
      )}
    </div>
  );
